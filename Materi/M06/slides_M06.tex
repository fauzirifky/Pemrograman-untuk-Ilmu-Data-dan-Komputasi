\begin{frame}{Masalah yang akan dikerjakan}
\begin{columns}[T]
\column{.49\textwidth}
\begin{block}{Searching}
Diberikan data dan satu nilai target. Kita perlu memutuskan:
\begin{itemize}
  \item target ada atau tidak;
  \item posisi target jika ditemukan;
  \item berapa banyak perbandingan yang diperlukan.
\end{itemize}
\end{block}
\column{.47\textwidth}
\begin{block}{Sorting}
Diberikan data acak. Kita perlu menyusun data menjadi terurut, lalu memahami:
\begin{itemize}
  \item keadaan data setelah setiap langkah;
  \item operasi yang paling sering terjadi;
  \item mengapa beberapa algoritma lebih cepat.
\end{itemize}
\end{block}
\end{columns}
\vspace{1mm}

\end{frame}

% 3
\begin{frame}[fragile]{Sequential search: memeriksa satu per satu}
\begin{columns}[T]
\column{.50\textwidth}
\begin{lstlisting}
def sequential_search(a, key):
    for i in range(len(a)):
        if a[i] == key:
            return i
    return -1
\end{lstlisting}
\begin{block}{Operasi utama}
Perbandingan \texttt{a[i] == key}. Banyaknya perbandingan bergantung posisi target.
\end{block}
\column{.46\textwidth}
\begin{center}
\begin{tikzpicture}[x=8mm,y=7mm]
\foreach \i/\v in {0/3,1/5,2/6,3/8,4/11,5/12,6/14,7/15,8/17,9/18}{
  \node[draw=Navy!40,fill=SoftGray,minimum width=7mm,minimum height=7mm] at (\i,0) {\scriptsize \v};
  \node[below] at (\i,-.55) {\tiny \i};
}
\draw[decorate,decoration={brace,amplitude=3pt},Navy,thick] (-.4,.65) -- (3.4,.65) node[midway,above=3pt]{\scriptsize key = 8};
\foreach \i in {0,1,2,3}{\draw[->,Warn,thick] (\i,.85) -- (\i,.35);}
\end{tikzpicture}
\end{center}
\vspace{1mm}
\begin{itemize}
  \item awal: $1$ perbandingan.
  \item tengah: sekitar $n/2$ perbandingan.
  \item akhir/tidak ada: $n$ perbandingan.
\end{itemize}
\[
T_{\text{worst}}(n)=n \quad \Rightarrow \quad O(n)
\]
\end{columns}
\end{frame}

\begin{frame}{Sequential search: empat posisi target}
\small
Gunakan list terurut berikut hanya untuk memudahkan pelacakan:
\[
[3,5,6,8,11,12,14,15,17,18]
\]
\begin{center}
\begin{tabular}{lccc}
\toprule
Kasus & Target & Banyak perbandingan & Orde terburuk \\
\midrule
Target di awal & $3$ & $1$ & $O(n)$ \\
Target di tengah & $12$ & $6$ & $O(n)$ \\
Target di akhir & $18$ & $10$ & $O(n)$ \\
Target tidak ada & $16$ & $10$ & $O(n)$ \\
\bottomrule
\end{tabular}
\end{center}
\vspace{2mm}
\begin{block}{Catatan penting}
Terurut atau tidak terurut, sequential search tetap memeriksa elemen satu per satu. Data terurut baru benar-benar dimanfaatkan oleh binary search.
\end{block}
\end{frame}

% 6
\begin{frame}{Mengapa binary search menghasilkan $O(\log n)$?}
\begin{columns}[T]
\column{.47\textwidth}
\begin{center}
\begin{tikzpicture}[node distance=5mm]
\node[box,minimum width=5.5cm] (a) {$n$ kandidat};
\node[box,minimum width=4.2cm,below=of a] (b) {$n/2$ kandidat};
\node[box,minimum width=3.1cm,below=of b] (c) {$n/4$ kandidat};
\node[box,minimum width=2.2cm,below=of c] (d) {$n/8$ kandidat};
\node[darkbox,minimum width=1.5cm,below=of d] (e) {$1$ kandidat};
\draw[arr] (a) -- (b); \draw[arr] (b) -- (c); \draw[arr] (c) -- (d); \draw[arr] (d) -- (e);
\end{tikzpicture}
\end{center}
\column{.49\textwidth}
Setelah $k$ langkah:
\[
\frac{n}{2^k}\le 1
\]
\[
n\le 2^k
\]
\[
k\ge \log_2 n
\]
\begin{block}{Orde}
\[
T(n)\approx \lfloor \log_2 n\rfloor+1 \quad \Rightarrow \quad O(\log n)
\]
\end{block}
\end{columns}
\end{frame}

% 7
\begin{frame}{Selection sort: satu pass mencari nilai terkecil}
\small
Data awal:
\[
[11,7,12,14,19,1,6,18,8,20]
\]
\begin{block}{Definisi satu pass}
Pada pass ke-$k$, cari nilai terkecil pada bagian belum terurut, lalu tukar ke posisi ke-$k$.
\end{block}
\begin{center}
\begin{tabular}{ccl}
\toprule
Pass & Nilai minimum bagian belum terurut & Keadaan list \\
\midrule
0 & -- & $[11,7,12,14,19,1,6,18,8,20]$ \\
1 & $1$ & $[1,7,12,14,19,11,6,18,8,20]$ \\
2 & $6$ & $[1,6,12,14,19,11,7,18,8,20]$ \\
3 & $7$ & $[1,6,7,14,19,11,12,18,8,20]$ \\
\bottomrule
\end{tabular}
\end{center}
\vspace{1mm}
\colorbox{LightTeal}{\parbox{.94\linewidth}{\small Setelah pass ke-$k$, $k$ elemen pertama sudah berada pada posisi final.}}
\end{frame}

% 8
\begin{frame}{Selection sort: dari invariant ke $T(n)$}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Invariant}
Setelah pass ke-$k$, bagian kiri sepanjang $k$ elemen berisi $k$ nilai terkecil dan sudah terurut.
\end{block}
\begin{center}
\begin{tikzpicture}[x=8mm,y=7mm]
\foreach \i/\v in {0/1,1/6,2/7,3/14,4/19,5/11,6/12,7/18,8/8,9/20}{
  \node[draw=Navy!35,fill=SoftGray,minimum width=7mm,minimum height=7mm] at (\i,0) {\scriptsize \v};
}
\draw[decorate,decoration={brace,amplitude=3pt},Good!80!black,thick] (-.4,.65) -- (2.4,.65) node[midway,above=3pt]{\scriptsize final};
\draw[decorate,decoration={brace,amplitude=3pt},Warn!90!black,thick] (2.6,-.65) -- (9.4,-.65) node[midway,below=3pt]{\scriptsize belum terurut};
\end{tikzpicture}
\end{center}
\column{.48\textwidth}
Banyak perbandingan untuk mencari minimum:
\[
(n-1)+(n-2)+\cdots+1
\]
\[
T(n)=\frac{n(n-1)}{2}=\frac{1}{2}n^2-\frac{1}{2}n
\]
\[
T(n)\in O(n^2)
\]
\begin{block}{Makna}
Selection sort tetap membandingkan hampir semua pasangan kandidat, walaupun swap hanya sekali per pass.
\end{block}
\end{columns}
\end{frame}

\begin{columns}[T]
\column{.53\textwidth}
\small
Data:
\[
[21,1,26,45,29,28,2,9,16,49,39,27,43,34,46,40]
\]
\begin{center}
\begin{tikzpicture}[level distance=9mm,sibling distance=22mm, every node/.style={font=\scriptsize,align=center}]

\column{.43\textwidth}
Tiga pemanggilan rekursif pada cabang kiri:

Sublist yang sedang diproses:

\]

\[
[21]\quad \text{dan}\quad [1]
\]
Hasil merge:

\]
\end{columns}
\end{frame}

% 10
\begin{frame}{Merge sort: mengapa $O(n\log n)$?}
\begin{columns}[T]
\column{.45\textwidth}
\begin{center}
\begin{tikzpicture}[x=1cm,y=.65cm]
\foreach \lvl/\cnt/\lab in {0/1/$n$,1/2/$n$,2/4/$n$,3/8/$n$}{
  \node[left] at (-.2,-\lvl) {\scriptsize level \lvl};
  \draw[Navy!60,thick] (0,-\lvl) -- (5,-\lvl);
  \node[right] at (5.15,-\lvl) {\scriptsize kerja total $\approx n$};
}
\draw[decorate,decoration={brace,amplitude=4pt},Warn,thick] (-.7,.2) -- (-.7,-3.2) node[midway,left=5pt]{\scriptsize $\log_2 n$ level};
\end{tikzpicture}
\end{center}
\column{.51\textwidth}
Pada setiap level, seluruh elemen ikut dalam proses merge.
\[
\text{kerja per level} \approx n
\]
Jumlah level pemecahan:
\[
\log_2 n
\]
Maka:
\[
T(n)\approx n\log_2 n \quad \Rightarrow \quad O(n\log n)
\]
\begin{block}{Perbandingan intuisi}
Selection sort membandingkan berulang pada bagian besar. Merge sort mengatur kerja dalam level-level pemecahan.
\end{block}
\end{columns}
\end{frame}

% 11
\begin{frame}{Sorting: algoritma berbeda, biaya berbeda}
\small
\begin{center}
\begin{tabular}{lll}
\toprule
Algoritma & Ide utama & Worst-case \\
\midrule
Bubble sort & Tukar elemen bertetangga yang salah urut & $O(n^2)$ \\
Selection sort & Pilih minimum, pindahkan ke depan & $O(n^2)$ \\
Insertion sort & Sisipkan elemen ke bagian kiri yang sudah terurut & $O(n^2)$ \\
Merge sort & Pecah dua, urutkan, lalu gabung & $O(n\log n)$ \\
Quicksort standar & Pilih pivot, partisi kiri-kanan & $O(n^2)$ \\
Python \texttt{sort} & Timsort, sangat kuat untuk data nyata & $O(n\log n)$ \\
\bottomrule
\end{tabular}
\end{center}
\vspace{2mm}
\begin{block}{Yang dijamin $O(n\log n)$ pada worst-case}
Dari daftar klasik di atas: merge sort. Python \texttt{sort} juga memiliki worst-case $O(n\log n)$ dalam implementasi modernnya.
\end{block}
\end{frame}

\begin{frame}[fragile]{Benchmark yang adil: bukan sekadar mengukur waktu}
\begin{columns}[T]
\column{.50\textwidth}
\begin{lstlisting}
for n in ukuran:
    data = buat_data_acak(n)
    for ulang in range(R):
        x = data.copy()
        ukur(insertion_sort, x)
        y = data.copy()
        ukur(merge_sort, y)
\end{lstlisting}
\column{.46\textwidth}
\begin{block}{Syarat penting}
\begin{itemize}
  \item ukuran input meningkat bertahap;
  \item kedua algoritma mendapat data yang sama;
  \item data harus di-\texttt{copy} sebelum diurutkan;
  \item pengukuran diulang, gunakan median/rata-rata;
  \item uji data acak, terurut, dan hampir terurut.
\end{itemize}
\end{block}
\end{columns}
\vspace{1mm}
\begin{block}{Membaca hasil}
Big-O menjelaskan pola asimtotik. Waktu aktual masih dipengaruhi komputer, implementasi, konstanta, dan bentuk data.
\end{block}
\end{frame}

% 13
\begin{frame}{Cek pemahaman}
\small
\begin{enumerate}
  \item Binary search pada $[3,5,6,8,11,12,14,15,17,18]$ dengan konvensi midpoint sublist. Urutan perbandingan untuk key $8$ dan key $16$ apa?
    \begin{itemize}\footnotesize
    \item Petunjuk: tulis sublist aktif setiap langkah, bukan hanya indeks awal.
  \end{itemize}
  \item Selection sort pada $[11,7,12,14,19,1,6,18,8,20]$. Keadaan list setelah tiga pass?
    \begin{itemize}\footnotesize
    \item Petunjuk: setiap pass memilih minimum dari bagian yang belum final.
  \end{itemize}
  \item Merge sort pada list 16 elemen. Setelah tiga pemanggilan rekursif kiri, sublist mana yang diproses?
    \begin{itemize}\footnotesize
    \item Petunjuk: ikuti cabang kiri: 16, 8, 4, 2.
  \end{itemize}
  \item Bagaimana menurunkan $T(n)$ selection sort dari jumlah perbandingan?
    \begin{itemize}\footnotesize
    \item Petunjuk: jumlahkan $(n-1)+(n-2)+\cdots+1$.
  \end{itemize}
\end{enumerate}
\end{frame}
