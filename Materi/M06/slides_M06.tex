% 3
\begin{frame}[fragile]{Searching I: sequential search}
\begin{columns}[T]
\column{.47\textwidth}
\begin{block}{Tujuan}
Cari target dalam list. Algoritma memeriksa elemen dari kiri ke kanan sampai ketemu atau sampai list habis.
\end{block}
\begin{lstlisting}
def sequential_search(a, target):
    for i in range(len(a)):
        if a[i] == target:
            return i
    return -1
\end{lstlisting}
\column{.47\textwidth}
\begin{block}{Banyak kerja: perbandingan}
Misalkan panjang list adalah $n$.
\[
\begin{array}{c|c|c}
\text{posisi target} & \text{contoh} & T(n)\\\hline
\text{awal} & a_0 & 1\\
\text{tengah} & a_{\lfloor n/2\rfloor} & \lfloor n/2\rfloor+1\\
\text{akhir} & a_{n-1} & n\\
\text{tidak ada} & - & n
\end{array}
\]
\end{block}
\pause
\begin{alertblock}{Worst case}
Untuk analisis dasar, sequential search biasanya dibaca $O(n)$.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Binary search: dari separuh ke $\log n$}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Ukuran kandidat setelah $k$ langkah}
\[
\begin{array}{c|c}
\text{langkah} & \text{sisa kandidat}\\\hline
0 & n\\
1 & n/2\\
2 & n/4\\
3 & n/8\\
k & n/2^k
\end{array}
\]
\end{block}
\column{.48\textwidth}
\begin{block}{Kapan berhenti?}
Kandidat tinggal satu ketika
\[
\frac{n}{2^k}\le 1.
\]
Maka
\[
 n \le 2^k \quad \Rightarrow \quad k \ge \log_2 n.
\]
\end{block}
\pause
\begin{alertblock}{Orde}
Binary search membutuhkan sekitar $\log_2 n$ perbandingan pada worst case, sehingga ordonya $O(\log n)$.
\end{alertblock}
\end{columns}
\end{frame}

% 7
\begin{frame}{Selection sort: menghitung $T(n)$}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Jumlah perbandingan}
Untuk mencari minimum:
\[
\begin{array}{c|c}
\text{pass} & \text{perbandingan}\\\hline
1 & n-1\\
2 & n-2\\
3 & n-3\\
\vdots & \vdots\\
n-1 & 1
\end{array}
\]
\end{block}
\column{.48\textwidth}
\begin{block}{Formulasi}
\[
\begin{aligned}
T(n)&=(n-1)+(n-2)+\cdots+1\\
&=1+2+\cdots+(n-1)\\
&=\frac{n(n-1)}{2}\\
&=\frac12 n^2-\frac12 n.
\end{aligned}
\]
\end{block}
\pause
\begin{alertblock}{Orde}
Suku dominan adalah $n^2$, jadi selection sort berada pada $O(n^2)$.
\end{alertblock}
\end{columns}
\vspace{1mm}
\small Bagian kiri yang sudah final tidak berubah lagi; dalam analisis algoritma, sifat tetap seperti ini biasa dikenal sebagai \textit{invariant}.
\end{frame}

\begin{frame}{Sorting II: bubble sort sebagai pembanding}
\begin{block}{Ide}
Bandingkan pasangan yang bersebelahan. Jika urutannya salah, tukar. Setelah satu lintasan, elemen terbesar bergerak ke ujung kanan.
\end{block}

\draw[arr,bend left=28] (.78*1,-.85) to node[above]{\scriptsize lanjut} (.78*2,-.85);
\end{tikzpicture}
\end{center}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Banyak kerja}
Worst case juga membentuk deret
\[
(n-1)+(n-2)+\cdots+1.
\]
\end{block}
\column{.48\textwidth}
\begin{alertblock}{Catatan}
Selection sort dan bubble sort sama-sama sederhana dan cenderung brute-force: banyak pasangan kandidat tetap diperiksa. Orde worst case keduanya $O(n^2)$.
\end{alertblock}
\end{columns}
\end{frame}

% 10
\begin{frame}{Merge sort: mengapa $O(n\log n)$?}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Jumlah level}
Setiap pemecahan membagi ukuran menjadi dua:
\[
n,\; n/2,\; n/4,\; \ldots,\; 1.
\]
Agar sampai ukuran 1:
\[
\frac{n}{2^k}=1 \Rightarrow k=\log_2 n.
\]
\end{block}
\column{.48\textwidth}
\begin{block}{Kerja per level}
Pada setiap level, proses merge membaca total sekitar $n$ elemen.
\[
T(n) \approx \underbrace{n+n+\cdots+n}_{\log_2 n\;\text{level}}
= n\log_2 n.
\]
\end{block}
\pause
\begin{alertblock}{Orde}
Merge sort memiliki orde $O(n\log n)$ karena ada $\log n$ level, dan tiap level memproses total $n$ elemen.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Membaca hasil algoritma: teori dan eksperimen}
\begin{columns}[T]
\column{.50\textwidth}
\begin{table}
\small
\begin{tabular}{lll}
\toprule
Algoritma & Ide utama & Worst case\\
\midrule

Merge sort & pecah dan gabung & $O(n\log n)$\\
\bottomrule
\end{tabular}

\begin{block}{Benchmark yang adil}
\begin{itemize}
\item data input sama untuk algoritma yang dibandingkan;
\item ukuran input meningkat bertahap;
\item pengukuran diulang;
\item laporkan median/rata-rata;
\item interpretasi tetap dikaitkan dengan $T(n)$.
\end{itemize}
\end{block}
\end{columns}
\vspace{1mm}
\small Waktu aktual dipengaruhi komputer, bahasa, dan implementasi; Big-O menjelaskan pola pertumbuhan saat $n$ membesar.
\end{frame}

\begin{frame}{Cek pemahaman}
\begin{enumerate}
\item Untuk list terurut $[3,5,6,8,11,12,14,15,17,18]$, telusuri binary search untuk target 8 dan target 16.
\pause
\begin{itemize}
\item Petunjuk: pada setiap langkah, catat indeks kiri, tengah, kanan, dan nilai tengah.
\end{itemize}
\item Lakukan tiga pass pertama selection sort pada $[11,7,12,14,19,1,6,18,8,20]$.
\pause
\begin{itemize}
\item Petunjuk: setelah pass ke-$k$, $k$ elemen terkecil sudah berada pada posisi final.
\end{itemize}
\item Jelaskan mengapa insertion/selection/bubble dapat tampak cepat pada data kecil, tetapi berbeda dari merge sort ketika data membesar.
\pause
\begin{itemize}
\item Petunjuk: bandingkan bentuk $n^2$ dengan $n\log n$, bukan hanya satu waktu pengukuran.
\end{itemize}
\end{enumerate}
\end{frame}
