\begin{frame}{Orde: dari ukuran biasa ke pertumbuhan fungsi}
\begin{columns}[T]
\column{.46\textwidth}
\begin{block}{Dalam bahasa sehari-hari}
\begin{itemize}
\item puluhan: kira-kira $10^1$
\item ratusan: kira-kira $10^2$
\item ribuan: kira-kira $10^3$
\end{itemize}
Kita tidak sedang mencari angka persis, tetapi kelas besarnya.
\end{block}
\column{.54\textwidth}
\begin{block}{Dalam algoritma}
Misalnya ukuran input adalah $n$.
\[
\begin{array}{c|c}
\text{kerja program} & \text{orde}\\
\hline
7n+3 & O(n)\\
5n^2+27n+1005 & O(n^2)\\
2n+26 & O(n)
\end{array}
\]
\end{block}
\end{columns}
\vspace{1mm}
\begin{alertblock}{Catatan matematika}
Secara matematis, $f(n)=O(g(n))$ berarti setelah $n$ cukup besar, $f(n)$ dapat dibatasi oleh konstanta kali $g(n)$.
\end{alertblock}
\end{frame}

% 4
\begin{frame}{Cara membaca $T(n)$}
\begin{block}{Jangan dibaca sebagai turunan}
Di sini $T(n)$ berarti fungsi jumlah kerja terhadap ukuran input $n$.
Kata yang lebih tepat: \textbf{merumuskan} atau \textbf{menghitung} $T(n)$.
\end{block}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Langkah kerja di papan}
\begin{enumerate}
\item Tentukan ukuran input $n$.
\item Tentukan operasi yang dihitung.
\item Hitung berapa kali operasi itu dieksekusi.
\item Tulis $T(n)$ secara jelas.
\item Ambil suku yang dominan untuk Big-O.
\end{enumerate}
\end{block}
\column{.52\textwidth}
\begin{block}{Apa yang konstan?}
\begin{itemize}
\item Assign awal seperti \texttt{s = 0}: konstan.
\item \texttt{print} satu kali: konstan.
\item Loop sebanyak $n$: bergantung $n$.
\item Loop di dalam loop: biasanya bergantung hasil kali banyak iterasi.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Contoh 1: tujuan yang sama, cara kerja berbeda}
\begin{columns}[T]
\column{.50\textwidth}
\begin{block}{Tujuan}
Hitung
\[
S(n)=1+2+\cdots+n.
\]
Dua algoritma di bawah menghasilkan nilai yang sama.
\end{block}
\begin{lstlisting}
def jumlah_loop(n):
    s = 0
    for i in range(1, n+1):
        s = s + i
    return s
\end{lstlisting}
\column{.50\textwidth}
\begin{block}{Rumus tertutup}
\[
1+2+\cdots+n = \frac{n(n+1)}{2}.
\]
\begin{lstlisting}
def jumlah_rumus(n):
    return n*(n+1)//2
\end{lstlisting}
\end{block}
\pause
\begin{alertblock}{Pertanyaan kelas}
Bagian mana yang diulang sebanyak $n$ kali?
\end{alertblock}
\end{columns}
\end{frame}

% 6
\begin{frame}{Contoh 1: menghitung $T(n)$}
\begin{columns}[T]
\column{.50\textwidth}
\begin{block}{Versi loop}
Hitung operasi utama: \texttt{s = s + i}.
\[
\begin{aligned}
T_{\text{loop}}(n) &= n \\
\Rightarrow \quad T_{\text{loop}}(n) &= O(n).
\end{aligned}
\]
Jika ikut menghitung inisialisasi dan return, misalnya
\[
T(n)=n+2,
\]
orde tetap $O(n)$.
\end{block}
\column{.50\textwidth}
\begin{block}{Versi rumus}
Jumlah operasi aritmatika tidak bertambah saat $n$ membesar.
\[
T_{\text{rumus}}(n)=c
\quad \Rightarrow \quad O(1).
\]
\end{block}
\pause
\begin{alertblock}{Pesan}
Dua program dapat menghasilkan jawaban yang sama, tetapi biaya komputasinya berbeda.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Contoh 2: mengapa membagi dua menjadi $\log n$?}
\begin{columns}[T]
\column{.46\textwidth}
\begin{lstlisting}
def bagi_dua(n):
    i = n
    while i > 0:
        i = i // 2
\end{lstlisting}
\column{.54\textwidth}
\begin{block}{Telusuri untuk $n=40$}
\[
40 \to 20 \to 10 \to 5 \to 2 \to 1 \to 0.
\]
\pause
Setelah $k$ kali:
\[
\frac{n}{2^k} < 1.
\]
Maka
\[
2^k > n \quad \Rightarrow \quad k > \log_2 n.
\]
\end{block}
\pause
\begin{alertblock}{Kesimpulan}
Jumlah iterasi bertambah lambat: $O(\log n)$.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Kasus utama: apakah dua kata anagram?}
\begin{block}{Definisi kerja}
Dua string disebut anagram jika keduanya memakai karakter yang sama dengan banyak kemunculan yang sama, hanya urutannya berbeda.
\end{block}
\begin{center}
\begin{tikzpicture}
\node[box] (a) {\texttt{heart}};
\node[box,right=2cm of a] (b) {\texttt{earth}};
\draw[arr] (a) -- node[above]{susun ulang} (b);
\end{tikzpicture}
\end{center}
\begin{block}{Tiga strategi yang akan dibandingkan}
\begin{enumerate}
\item Cek satu per satu dan tandai yang sudah dipakai.
\item Urutkan dua string, lalu bandingkan.
\item Hitung frekuensi setiap huruf, lalu bandingkan penghitungnya.
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}[fragile]{Anagram A: cek dan tandai}
\begin{columns}[T]
\column{.45\textwidth}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
def anagram_A(s1, s2):
    data = list(s2)
    for ch in s1:
        ketemu = False
        for j in range(len(data)):
            if ch == data[j]:
                data[j] = None
                ketemu = True
                break
        if not ketemu:
            return False
    return True
\end{lstlisting}

\begin{frame}{Anagram A: dari hitungan ke $O(n^2)$}
\begin{block}{Ukuran input}
Misalkan panjang string adalah $n$.
\end{block}
\begin{columns}[T]
\column{.50\textwidth}
\begin{block}{Pada kasus yang berat}
\begin{itemize}
\item Huruf pertama bisa memeriksa sampai $n$ posisi.
\item Huruf kedua bisa memeriksa sampai $n-1$ posisi yang tersisa.
\item Dan seterusnya.
\end{itemize}
\end{block}
\column{.50\textwidth}
\[
T(n) = n+(n-1)+\cdots+1.
\]
\pause
\[
T(n)=\frac{n(n+1)}{2}=\frac{1}{2}n^2+\frac{1}{2}n.
\]
\pause
\[
T(n)=O(n^2).
\]
\end{columns}
\end{frame}

    return a == b
\end{lstlisting}
\begin{center}
\texttt{heart} $\to$ \texttt{aehrt}\qquad
\texttt{earth} $\to$ \texttt{aehrt}
\end{center}
\column{.52\textwidth}
\begin{block}{Hitung bagian kerjanya}
\begin{itemize}
\item Mengurutkan string pertama.
\item Mengurutkan string kedua.
\item Membandingkan hasil urutan.
\end{itemize}
\end{block}
\pause
Jika sorting yang dipakai $O(n\log n)$, maka
\[
T(n) = 2\,O(n\log n) + O(n).
\]
\pause
Suku dominan: $O(n\log n)$.
\end{columns}
\end{frame}

\begin{frame}[fragile]{Anagram C: hitung frekuensi}
\begin{columns}[T]
\column{.44\textwidth}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
def anagram_C(s1, s2):
    c1 = [0]*26
    c2 = [0]*26
    for ch in s1:
        c1[ord(ch)-ord('a')] += 1
    for ch in s2:
        c2[ord(ch)-ord('a')] += 1
    return c1 == c2
\end{lstlisting}
\column{.56\textwidth}
\begin{center}

Karakter tidak dicari satu per satu. Setiap karakter langsung menambah penghitungnya.
\end{block}

% 21
\begin{frame}{Anagram C: mengapa $O(n)$?}
\begin{columns}[T]
\column{.50\textwidth}
\begin{block}{Bagian yang dihitung}
\begin{itemize}
\item Menyapu \texttt{s1}: $n$ langkah.
\item Menyapu \texttt{s2}: $n$ langkah.
\item Membandingkan 26 penghitung: 26 langkah.
\end{itemize}
\end{block}
\column{.50\textwidth}
\[
T(n)=n+n+26=2n+26.
\]
\pause
\[
T(n)=O(n).
\]
\pause
\begin{alertblock}{Trade-off}
Lebih cepat, tetapi memakai memori tambahan untuk penghitung.
\end{alertblock}
\end{columns}
\end{frame}

% 23
\begin{frame}[fragile]{Self check: baca kode, bukan tebak notasi}
\begin{lstlisting}
test = 0
for i in range(n):
    for j in range(n):
        test = test + i*j
\end{lstlisting}
\pause
\begin{block}{Petunjuk}
Untuk setiap nilai $i$, loop $j$ berjalan $n$ kali. Ada $n$ pilihan $i$.
\end{block}
\pause
\[
T(n)=n\cdot n=n^2 \quad \Rightarrow \quad O(n^2).
\]
\end{frame}

% 24
\begin{frame}[fragile]{Self check: dua loop berurutan}
\begin{lstlisting}
test = 0
for i in range(n):
    test = test + 1
for j in range(n):
    test = test - 1
\end{lstlisting}
\pause
\begin{block}{Petunjuk}
Loop kedua tidak berada di dalam loop pertama. Kerjanya dijumlahkan, bukan dikalikan.
\end{block}
\pause
\[
T(n)=n+n=2n \quad \Rightarrow \quad O(n).
\]
\end{frame}

% 25
\begin{frame}[fragile]{Self check: dibagi dua}
\begin{lstlisting}
i = n
while i > 0:
    k = 2 + 2
    i = i // 2
\end{lstlisting}
\pause
\begin{block}{Petunjuk}
Tulis nilai $i$ setelah setiap iterasi: $n, n/2, n/4, n/8, \ldots$.
\end{block}
\pause
Berhenti ketika $n/2^k < 1$, sehingga $k > \log_2 n$.
\[
T(n)=O(\log n).
\]
\end{frame}
