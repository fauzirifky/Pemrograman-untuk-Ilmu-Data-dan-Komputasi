% 3
\begin{frame}[fragile]{Vektor: indeks, operasi komponen, dan norma}
\begin{columns}[T]
\column{.49\textwidth}
\begin{block}{Dari rumus ke indeks}
\small
\[
x=(x_1,\ldots,x_n),\qquad y=(y_1,\ldots,y_n)
\]
\[
x+y=(x_i+y_i)_{i=1}^{n},\quad
x\cdot y=\sum_{i=1}^{n}x_i y_i
\]
\[
\|x-y\|_2=\sqrt{\sum_{i=1}^{n}(x_i-y_i)^2}
\]
\end{block}
\pause
\begin{block}{Fungsi kerja}
Jika loop berjalan pada $n$ komponen, maka banyak kerja bertambah sebanding dengan $n$.
\[
W(n)=\text{kerja tetap}+\text{kerja per komponen}\cdot n
\]
\[
W(n)\in O(n)
\]
\end{block}
\column{.47\textwidth}
\begin{lstlisting}
def dot(x, y):
    if len(x) != len(y):
        raise ValueError("dimensi beda")
    total = 0
    for i in range(len(x)):
        total += x[i] * y[i]
    return total

def norm_diff(x, y):
    return dot([x[i]-y[i] for i in range(len(x))],
               [x[i]-y[i] for i in range(len(x))]) ** 0.5
\end{lstlisting}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Element-wise dan perkalian matriks}
\begin{columns}[T]
\column{.47\textwidth}
\[
A=\mat{1&2\\3&4},\qquad
B=\mat{2&0\\1&5}
\]
\[
A\odot B=\mat{1\cdot2&2\cdot0\\3\cdot1&4\cdot5}
=\mat{2&0\\3&20}
\]
\[
AB=\mat{
1\cdot2+2\cdot1 & 1\cdot0+2\cdot5\\
3\cdot2+4\cdot1 & 3\cdot0+4\cdot5}
=\mat{4&10\\10&20}
\]
\column{.49\textwidth}
\begin{lstlisting}
import numpy as np
A = np.array([[1, 2], [3, 4]])
B = np.array([[2, 0], [1, 5]])

A * B   # element-wise
A @ B   # matrix product
\end{lstlisting}
\pause
\begin{alertblock}{Catatan}
\texttt{*} membaca posisi yang sama. \texttt{@} membaca baris pada $A$ dan kolom pada $B$.
\end{alertblock}
\end{columns}
\end{frame}

% 5
\begin{frame}[fragile]{Perkalian matriks: pseudocode dulu, kode kemudian}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Pseudocode}
\scriptsize
\begin{verbatim}
INPUT: A berukuran m x p, B berukuran p x n
OUTPUT: C = A B berukuran m x n

cek banyak kolom A = banyak baris B
buat C berisi nol, ukuran m x n
FOR i = 0 sampai m-1:
  FOR j = 0 sampai n-1:
    s = 0
    FOR k = 0 sampai p-1:
      s = s + A[i][k] * B[k][j]
    C[i][j] = s
RETURN C
\end{verbatim}
\end{block}
\pause
\column{.48\textwidth}
\begin{lstlisting}
def matmul_manual(A, B):
    m = len(A)
    p = len(A[0])
    p2 = len(B)
    n = len(B[0])
    if p != p2:
        raise ValueError("dimensi tidak cocok")

    C = [[0 for j in range(n)] for i in range(m)]
    for i in range(m):
        for j in range(n):
            s = 0
            for k in range(p):
                s += A[i][k] * B[k][j]
            C[i][j] = s
    return C
\end{lstlisting}
\end{columns}
\end{frame}

% 6
\begin{frame}{Fungsi kerja dari kode perkalian matriks}
Kita hitung dari struktur kode, bukan dari nama algoritmanya.
\vspace{2mm}
\begin{center}
\begin{tabular}{p{7.4cm}r}
\toprule
Bagian kode & Banyak kerja pokok \\
\midrule
\texttt{m=len(A)}, \texttt{p=len(A[0])}, \texttt{p2=len(B)}, \texttt{n=len(B[0])} & $4$ \\
\texttt{if p != p2} & $1$ perbandingan \\
Membuat matriks \texttt{C} berisi nol & $mn$ penulisan \\
\texttt{s = 0} untuk setiap pasangan $(i,j)$ & $mn$ penulisan \\
\texttt{s += A[i][k] * B[k][j]} & $mnp$ kali; tiap kali $1$ kali dan $1$ tambah \\
\texttt{C[i][j] = s} & $mn$ penulisan \\
\bottomrule
\end{tabular}
\end{center}
\pause
\[
W(m,p,n)=5+mn+mn+2mnp+mn=5+3mn+2mnp.
\]
Jika matriks bujur sangkar $n\times n$:
\[
W(n)=5+3n^2+2n^3 \quad\Rightarrow\quad W(n)\in O(n^3).
\]
\end{frame}

% 7
\begin{frame}[fragile]{Transpose: algoritma sederhana, biaya linear terhadap jumlah elemen}
\begin{columns}[T]
\column{.42\textwidth}
\[
A=\begin{pmatrix}
1&2&3\\
4&5&6
\end{pmatrix}
\quad\Rightarrow\quad
A^T=\begin{pmatrix}
1&4\\2&5\\3&6
\end{pmatrix}
\]
\begin{block}{Pseudocode}
\scriptsize
\begin{verbatim}
INPUT: A berukuran m x n
buat T berukuran n x m
FOR i = 0..m-1:
  FOR j = 0..n-1:
    T[j][i] = A[i][j]
RETURN T
\end{verbatim}
\end{block}
\pause
\column{.54\textwidth}
\begin{lstlisting}
def transpose(A):
    m = len(A)
    n = len(A[0])
    T = [[0 for i in range(m)] for j in range(n)]
    for i in range(m):
        for j in range(n):
            T[j][i] = A[i][j]
    return T

# NumPy:
# np.array(A).T
\end{lstlisting}
\begin{alertblock}{Fungsi kerja}
Ada $mn$ elemen yang dipindahkan. Maka $W(m,n)$ sebanding dengan $mn$.
\end{alertblock}
\end{columns}
\end{frame}

% 8
\begin{frame}[fragile]{Invers 2x2: rumus, domain, dan pemeriksaan}
\begin{columns}[T]
\column{.44\textwidth}
\[
A=\mat{a&b\\c&d},\qquad
A^{-1}=\frac{1}{ad-bc}\mat{d&-b\\-c&a}
\]
\begin{alertblock}{Syarat}
Invers ada jika dan hanya jika
\[
ad-bc\ne 0.
\]
Jika determinan nol, program harus menolak input.
\end{alertblock}
\column{.52\textwidth}
\begin{lstlisting}
def inv2(A):
    a, b = A[0]
    c, d = A[1]
    det = a*d - b*c
    if det == 0:
        raise ValueError("matriks singular")
    return [[ d/det, -b/det],
            [-c/det,  a/det]]
\end{lstlisting}
\pause
\begin{block}{Verifikasi}
Hasil invers tidak cukup dilihat dari bentuknya. Periksa
\[
AA^{-1}\approx I.
\]
Pada komputer, gunakan toleransi, bukan kesamaan desimal mentah.
\end{block}
\end{columns}
\end{frame}

% 9
\begin{frame}[fragile]{SPL: selesaikan, lalu ukur residual}
\begin{columns}[T]
\column{.46\textwidth}
Untuk $Ax=b$, solusi numerik $\hat{x}$ sebaiknya diperiksa dengan residual:
\[
r=b-A\hat{x}.
\]
Jika $\|r\|_2$ kecil, solusi konsisten dengan sistem.
\begin{block}{Mengapa residual?}
\small
Kesalahan kecil pada komponen $x$ dapat sulit dilihat langsung, tetapi $A\hat{x}$ harus tetap dekat dengan $b$.
\end{block}
\column{.50\textwidth}
\begin{lstlisting}
import numpy as np

A = np.array([[2, 1], [5, 3]], dtype=float)
b = np.array([1, 4], dtype=float)

x = np.linalg.solve(A, b)
r = b - A @ x
err = np.linalg.norm(r)
print(x, err)
\end{lstlisting}
\pause
\begin{alertblock}{Praktik komputasi}
Untuk menyelesaikan $Ax=b$, gunakan \texttt{solve}. Jangan membentuk $A^{-1}$ hanya untuk mengalikan $A^{-1}b$.
\end{alertblock}
\end{columns}
\end{frame}

% 10
\begin{frame}{Pengayaan: invers ukuran $n\times n$ dan biaya komputasi}
\begin{block}{Satu cara berpikir}
Untuk matriks besar, invers dapat dipandang sebagai menyelesaikan
\[
AX=I.
\]
Kolom ke-$j$ dari $X$ diperoleh dari
\[
Ax_j=e_j.
\]
Artinya, menghitung invers berarti menyelesaikan banyak sistem linear.
\end{block}
\pause
\begin{center}
\begin{tikzpicture}[node distance=6mm]
\node[box,text width=2.8cm] (a) {Eliminasi Gaussian / faktorisasi};
\node[box,text width=2.5cm,right=of a] (b) {banyak operasi bertingkat};
\node[box,text width=2.4cm,right=of b] (c) {orde kubik};
\draw[arr] (a) -- (b);
\draw[arr] (b) -- (c);
\end{tikzpicture}
\end{center}
\[
W(n)\in O(n^3),\qquad \text{memori matriks }\in O(n^2).
\]
\begin{alertblock}{Kaitannya dengan data besar}
Jika $n$ ribuan atau lebih, membentuk invers eksplisit sering mahal dan tidak stabil. Pilih algoritma yang langsung menjawab kebutuhan: solve, least squares, sparse solver, atau metode iteratif.
\end{alertblock}
\end{frame}

% 11
\begin{frame}{Eigenvalue: lebih dari sekadar menghitung akar}
\begin{columns}[T]
\column{.47\textwidth}
\begin{block}{Definisi}
\[
Av=\lambda v,
\qquad v\ne 0.
\]
Vektor $v$ hanya berubah skala oleh transformasi $A$.
\end{block}
\begin{block}{Persamaan karakteristik}
\[
\det(A-\lambda I)=0.
\]
Untuk matriks kecil, ini masih dapat dihitung manual.
\end{block}
\column{.49\textwidth}
\begin{alertblock}{Teorema yang berguna}
Untuk matriks bujur sangkar $A$:
\begin{itemize}\small
  \item $A$ invertible $\Longleftrightarrow \det(A)\ne0$.
  \item $A$ invertible $\Longleftrightarrow 0$ bukan eigenvalue $A$.
  \item Jika eigenvalue $A$ adalah $\lambda_i$, maka
  \[
  \det(A)=\prod_i \lambda_i.
  \]
  \item Jika $A$ invertible, eigenvalue $A^{-1}$ adalah $1/\lambda_i$.
\end{itemize}
\end{alertblock}
\end{columns}
\end{frame}

% 12
\begin{frame}[fragile]{Eigen di Python: hitung dan verifikasi}
\begin{columns}[T]
\column{.42\textwidth}
\[
A=\mat{2&1\\1&2}
\]
\[
\det(A-\lambda I)=
\det\mat{2-\lambda&1\\1&2-\lambda}
\]
\[
=(2-\lambda)^2-1=0
\]
\[
\lambda=1,3.
\]
\column{.54\textwidth}
\begin{lstlisting}
import numpy as np

A = np.array([[2, 1], [1, 2]], dtype=float)
w, V = np.linalg.eig(A)

for k in range(len(w)):
    lam = w[k]
    v = V[:, k]
    cek = np.linalg.norm(A @ v - lam * v)
    print(lam, cek)
\end{lstlisting}
\pause
\begin{block}{Makna pemeriksaan}
Jika $\|Av-\lambda v\|_2$ kecil, pasangan $(\lambda,v)$ konsisten secara numerik.
\end{block}
\end{columns}
\end{frame}

% 13
\begin{frame}{Data sebagai matriks dan pintu masuk PCA}
\begin{columns}[T]
\column{.46\textwidth}
\begin{block}{Data}
Baris adalah observasi, kolom adalah fitur.
\[
X=\begin{pmatrix}
1&2&0\\
2&1&1\\
4&3&2\\
3&4&1
\end{pmatrix}
\]
Rata-rata kolom:
\[
\bar{x}=\frac{1}{m}\sum_{i=1}^{m}X_{i,:}.
\]
Centered matrix:
\[
X_c=X-\mathbf{1}\bar{x}^{T}.
\]
\end{block}
\column{.50\textwidth}
\begin{block}{Gram / covariance direction}
\[
G=X_c^T X_c.
\]
Matriks ini simetris dan memuat hubungan antarfitur.
\end{block}
\pause
\begin{alertblock}{Ide PCA}
PCA mencari arah baru yang menjelaskan variasi data sebesar mungkin. Secara aljabar linear, arah ini diperoleh dari eigenvector matriks kovariansi atau Gram yang sesuai.
\end{alertblock}
\end{columns}
\end{frame}

% 14
\begin{frame}[fragile]{Algoritma data sebagai matriks}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Pseudocode}
\scriptsize
\begin{verbatim}
INPUT: X berukuran m x n
hitung mean setiap kolom
buat Xc = X - mean_kolom
hitung G = Xc^T Xc
cek G simetris
ambil eigenvalue/eigenvector G jika perlu
\end{verbatim}
\end{block}
\pause
\column{.48\textwidth}
\begin{lstlisting}
import numpy as np

X = np.array([[1,2,0], [2,1,1],
              [4,3,2], [3,4,1]], dtype=float)

xbar = X.mean(axis=0)
Xc = X - xbar
G = Xc.T @ Xc
print(G)
print(np.allclose(G, G.T))
\end{lstlisting}
\end{columns}
\vspace{1mm}
\begin{alertblock}{Fungsi kerja}
Menghitung $X_c^T X_c$ untuk $X_c$ berukuran $m\times n$ memerlukan pola seperti perkalian matriks: $n\times m$ dikali $m\times n$, orde utamanya $O(mn^2)$.
\end{alertblock}
\end{frame}

\begin{frame}{Latihan penutup yang layak menjadi soal ujian}

Kerjakan berurutan:

\end{enumerate}
\end{block}
\begin{alertblock}{Target berpikir}
Bukan sekadar memakai fungsi pustaka, tetapi memahami apa yang dihitung, bagaimana mengecek hasilnya, dan bagaimana biayanya bertumbuh.
\end{alertblock}
