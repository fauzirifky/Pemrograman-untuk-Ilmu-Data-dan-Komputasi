\begin{frame}{Masalah Pengelompokan Data}
\begin{columns}[T]
\column{.51\textwidth}
\small
Pada KNN sebelumnya, setiap observasi memiliki \textbf{label kelas}. Pada pengelompokan, label tersebut tidak tersedia.

\vspace{2mm}
\begin{block}{Masalah komputasi}
Diberikan sekumpulan observasi
\[
x_1,x_2,\ldots,x_n\in\mathbb{R}^d,
\]
bagilah observasi ke beberapa kelompok sehingga objek dalam kelompok yang sama relatif lebih mirip.
\end{block}

\begin{alertblock}{Perbedaan utama}
\footnotesize
Algoritma tidak diberi jawaban ``kelas A'' atau ``kelas B''. Struktur kelompok dicari dari posisi dan kedekatan data itu sendiri.
\end{alertblock}

\begin{frame}{Representasi Matematis Pengelompokan}
\small
Untuk $n$ observasi dan $d$ fitur,
\[
X=
\begin{pmatrix}
x_1^T\\
x_2^T\\
\vdots\\
x_n^T
\end{pmatrix}
\in\mathbb{R}^{n\times d}.
\]

Kita ingin membentuk $k$ himpunan cluster
\[
C_1,C_2,\ldots,C_k
\]
dengan
\[
C_r\cap C_s=\varnothing\quad(r\neq s),
\qquad
C_1\cup C_2\cup\cdots\cup C_k=\{1,\ldots,n\}.
\]

\begin{block}{Makna indeks}
\footnotesize
Jika $i\in C_j$, observasi $x_i$ ditempatkan pada cluster ke-$j$. Nomor cluster hanyalah penanda; ``cluster 1'' tidak otomatis lebih baik atau lebih tinggi daripada ``cluster 2''.
\end{block}
\end{frame}

\small
\begin{center}
\begin{tabular}{p{3.0cm} p{5.0cm} p{5.0cm}}
\toprule
 & KNN & k-Means \\
\midrule
Jenis masalah & klasifikasi & pengelompokan \\
Data awal & memiliki label $y_i$ & tidak memiliki label kelas \\
Tujuan & memprediksi kelas objek baru & membentuk kelompok dari data \\
Peran jarak & mencari tetangga terdekat & menentukan centroid terdekat \\
Arti $k$ & banyak tetangga & banyak cluster \\
Hasil utama & label prediksi & anggota cluster dan centroid \\
\bottomrule
\end{tabular}
\end{center}

\small
Huruf $k$ pada KNN dan k-Means tidak menyatakan hal yang sama. Pada k-Means, $k$ harus ditentukan sebelum algoritma berjalan.
\end{alertblock}
\end{frame}

% 5
\begin{frame}{Centroid sebagai Pusat Cluster}
\begin{columns}[T]
\column{.50\textwidth}
\small
Misalkan cluster $C_j$ memiliki $n_j$ observasi. Centroid cluster tersebut adalah rata-rata vektor:
\[
\mu_j
=
\frac{1}{n_j}
\sum_{i\in C_j}x_i.
\]

Untuk data dua dimensi,
\[
\mu_j=
\begin{pmatrix}
\bar x_1\\
\bar x_2
\end{pmatrix},
\]
jadi koordinat centroid adalah rata-rata pada setiap fitur.

\begin{block}{Hubungan dengan aljabar linear}
\footnotesize
Centroid tetap merupakan vektor di $\mathbb{R}^d$. Operasi rata-rata dilakukan komponen demi komponen.
\end{block}

\column{.46\textwidth}
\centering
\begin{tikzpicture}[x=.72cm,y=.62cm]
  \draw[->,thick] (0,0) -- (6.2,0) node[right]{\scriptsize $x_1$};

  \foreach \p in {(1.1,1.1),(1.5,2.0),(2.2,1.3),(2.5,2.3),(3.0,1.7)}

  \fill[Gold!80!black] (2.06,1.68) circle (4pt);
  \node[anchor=west] at (2.2,1.68){\scriptsize centroid $\mu$};
\end{tikzpicture}
\end{columns}
\sumber{MacQueen (1967); Hastie, Tibshirani, \& Friedman (2009), \S14.3.6.}
\end{frame}

% 6
\begin{frame}{Mengapa Centroid Berupa Rata-Rata?}
\footnotesize
Untuk satu dimensi, misalkan kita mencari satu titik $m$ yang meminimumkan jumlah kuadrat jarak:
\[
f(m)=\sum_{i=1}^{n}(x_i-m)^2.
\]

Dengan kalkulus,
\[
f'(m)
=
-2\sum_{i=1}^{n}(x_i-m).
\]
Syarat titik stasioner $f'(m)=0$ memberi
\[
\sum_{i=1}^{n}x_i-nm=0
\quad\Longrightarrow\quad
m=\frac{1}{n}\sum_{i=1}^{n}x_i=\bar x.
\]

\begin{block}{Generalisasi}
\scriptsize
Pada $\mathbb{R}^d$, argumen yang sama berlaku untuk setiap koordinat. Karena itu pusat yang meminimumkan jumlah \textbf{kuadrat jarak Euclidean} adalah rata-rata vektor.
\end{block}
\sumber{Lloyd (1982), \textit{IEEE Transactions on Information Theory}, 28(2), 129--137.}
\end{frame}

% 7
\begin{frame}{Fungsi Objektif k-Means}
\footnotesize
Diberikan centroid $\mu_1,\ldots,\mu_k$. k-Means meminimumkan
\[
J
=
\sum_{j=1}^{k}
\sum_{i\in C_j}
\|x_i-\mu_j\|_2^2.
\]

\begin{center}
\begin{tabular}{p{2.2cm} p{10.5cm}}
\toprule
Notasi & Makna \\
\midrule
$x_i$ & observasi ke-$i$ \\
$C_j$ & himpunan indeks observasi dalam cluster ke-$j$ \\
$\mu_j$ & centroid cluster ke-$j$ \\
$\|x_i-\mu_j\|_2^2$ & kuadrat jarak observasi ke centroidnya \\
$J$ & total variasi di dalam cluster, sering disebut \textit{within-cluster sum of squares} \\
\bottomrule
\end{tabular}
\end{center}

\begin{alertblock}{Tujuan}
\scriptsize
Cluster yang baik menurut k-Means memiliki nilai $J$ kecil: observasi dekat dengan centroid kelompoknya.
\end{alertblock}
\sumber{Lloyd (1982); Hastie, Tibshirani, \& Friedman (2009), \S14.3.6.}
\end{frame}

% 8
\begin{frame}{Tahap Penetapan Keanggotaan Cluster (\textit{Assignment})}
\small
Pada tahap ini \textbf{centroid belum diubah}. Tugasnya hanya menentukan:
\[
\text{setiap observasi }x_i\text{ menjadi anggota cluster yang mana?}
\]

Dengan centroid sementara $\mu_1,\ldots,\mu_k$, hitung jarak
\[
d_{ij}=\|x_i-\mu_j\|_2,
\qquad j=1,\ldots,k.
\]
Kemudian pilih centroid yang paling dekat:
\[
c_i=\operatorname*{arg\,min}_{1\le j\le k} d_{ij}.
\]

\begin{columns}[T]
\column{.49\textwidth}
\begin{block}{Contoh}
\footnotesize
Jika
\[
(d_{i1},d_{i2},d_{i3})=(4.2,1.7,3.1),
\]
maka jarak terkecil adalah $1.7$ dan indeksnya $2$. Karena itu $x_i$ ditetapkan sebagai anggota cluster 2.
\end{block}
\column{.47\textwidth}
\begin{alertblock}{Yang berubah pada tahap ini}
\footnotesize
\textbf{Keanggotaan cluster berubah.}\\
Posisi centroid $\mu_j$ tetap. Centroid baru baru dihitung pada tahap \textit{update}.
\end{alertblock}
\end{columns}
\sumber{Lloyd (1982), kondisi nearest-centroid; Hastie et al. (2009), \S14.3.6.}
\end{frame}

% 9
\begin{frame}{Tahap Update Centroid}
\small
Setelah semua observasi mempunyai cluster, hitung ulang centroid setiap cluster:
\[
\mu_j^{\text{baru}}
=
\frac{1}{|C_j|}
\sum_{i\in C_j}x_i.
\]

\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Contoh}
\footnotesize
Jika
\[
C_1=\{(1,1),(2,1),(2,2)\},
\]
maka
\[
\mu_1
=
\left(
\frac{1+2+2}{3},
\frac{1+1+2}{3}
\right)
=
\left(\frac53,\frac43\right).
\]
\end{block}
\column{.48\textwidth}
\begin{block}{Interpretasi}
\footnotesize
Tahap \textit{assignment} menentukan \textbf{anggota cluster}. Tahap \textit{update} menghitung \textbf{centroid baru} dari anggota tersebut. Kedua tahap diulang sampai tidak terjadi perubahan berarti.
\end{block}
\end{columns}
\sumber{Lloyd (1982); MacQueen (1967).}
\end{frame}

% 10
\begin{frame}{Algoritma k-Means dengan Skema Lloyd}
\small
\begin{block}{Input}
Data $X\in\mathbb{R}^{n\times d}$, banyak cluster $k$, serta $k$ centroid awal
$\mu_1^{(0)},\ldots,\mu_k^{(0)}$.
\end{block}

\begin{enumerate}
\item \textbf{Inisialisasi:} tentukan $k$ centroid awal.
\item \textbf{Assignment:} untuk setiap $x_i$, pilih centroid terdekat dan tetapkan keanggotaan cluster.
\item \textbf{Update:} untuk setiap cluster, hitung centroid baru sebagai rata-rata anggotanya.
\item Hitung perubahan centroid atau periksa apakah keanggotaan cluster berubah.
\item Ulangi langkah 2--4 sampai konvergen atau mencapai batas iterasi.
\end{enumerate}

\begin{alertblock}{Sifat penting}
\footnotesize
Setiap siklus assignment--update tidak menaikkan fungsi objektif $J$. Namun solusi akhir dapat berupa minimum lokal dan bergantung pada inisialisasi.
\end{alertblock}
\sumber{Lloyd, S. (1982), \textit{Least squares quantization in PCM}, IEEE TIT 28(2), 129--137.}
\end{frame}

% 11
\begin{frame}{Beberapa Algoritma k-Means dan Strategi Inisialisasi}
\footnotesize
\begin{center}
\begin{tabular}{p{2.6cm} p{5.1cm} p{5.4cm}}
\toprule
Metode & Gagasan utama & Perbedaan singkat \\
\midrule
Lloyd & assignment seluruh data, lalu update seluruh centroid & pendekatan \textit{batch}; paling umum untuk pengantar \\
MacQueen & observasi diproses berurutan dan centroid dapat diperbarui segera & bersifat \textit{online}/sekuensial \\
Hartigan--Wong & mencoba memindahkan observasi antarcluster jika menurunkan within-cluster sum of squares & relokasi lokal lebih eksplisit \\
k-means++ & memilih centroid awal secara probabilistik agar tersebar & \textbf{inisialisasi}, bukan pengganti tahap assignment--update \\
\bottomrule
\end{tabular}
\end{center}

\begin{block}{Pada perkuliahan ini}
\small
Perhitungan manual dan implementasi utama menggunakan \textbf{skema Lloyd}. Metode lain dikenalkan untuk menunjukkan bahwa ``k-Means'' memiliki beberapa prosedur optimasi dan strategi inisialisasi.
\end{block}
\sumber{MacQueen (1967); Hartigan \& Wong (1979); Arthur \& Vassilvitskii (2007).}
\end{frame}

\begin{frame}{Cluster Awal: Hasil Assignment Pertama}
\scriptsize
Dengan centroid awal $\mu_1^{(0)}=(1,1)$ dan $\mu_2^{(0)}=(8,6)$, setiap observasi ditempatkan pada centroid terdekat.

\[
d((2,2),\mu_1^{(0)})=\sqrt2
<
d((2,2),\mu_2^{(0)})=\sqrt{52},
\]
maka $(2,2)$ menjadi anggota cluster 1.

\begin{center}
\begin{tabular}{c c c c}
\toprule
Titik & $d(\cdot,\mu_1^{(0)})$ & $d(\cdot,\mu_2^{(0)})$ & Cluster \\
\midrule
$(1,1)$ & $0$ & $\sqrt{74}$ & 1\\
$(1,2)$ & $1$ & $\sqrt{65}$ & 1\\
$(2,1)$ & $1$ & $\sqrt{61}$ & 1\\
$(2,2)$ & $\sqrt2$ & $\sqrt{52}$ & 1\\
$(7,5)$ & $\sqrt{52}$ & $\sqrt2$ & 2\\
$(8,5)$ & $\sqrt{65}$ & $1$ & 2\\

\end{center}

\qquad
\boxed{C_2^{(0)}=\{5,6,7,8\}}.

\end{frame}

\begin{frame}{Update Centroid dan Iterasi Berikutnya}
\small
Hasil assignment pertama:
\[
C_1=\{(1,1),(1,2),(2,1),(2,2)\},
\]
\[
C_2=\{(7,5),(8,5),(7,6),(8,6)\}.
\]

Centroid baru:
\[
\mu_1^{(1)}
=
\left(
\frac{1+1+2+2}{4},
\frac{1+2+1+2}{4}
\right)
=
(1.5,1.5),
\]
\[
\mu_2^{(1)}
=
\left(
\frac{7+8+7+8}{4},
\frac{5+5+6+6}{4}
\right)
=
(7.5,5.5).
\]

\begin{block}{Iterasi berikutnya}
\small
Dengan dua centroid baru ini, assignment tidak berubah. Maka algoritma telah konvergen untuk inisialisasi tersebut.
\end{block}
\sumber{Lloyd (1982).}
\end{frame}

% 14
\begin{frame}{Pengaruh Inisialisasi Centroid}
\small
k-Means tidak selalu menghasilkan solusi akhir yang sama untuk centroid awal yang berbeda.

\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Masalah}
\footnotesize
Fungsi objektif k-Means tidak selalu memiliki satu minimum yang mudah ditemukan. Algoritma Lloyd dapat berhenti pada \textbf{minimum lokal}.
\end{block}

\begin{block}{Praktik komputasi}
\footnotesize
Jalankan algoritma beberapa kali dengan centroid awal berbeda, lalu pilih hasil dengan nilai $J$ terkecil.
\end{block}

% 15
\begin{frame}{Normalisasi Fitur sebelum k-Means}
\small
Karena k-Means berbasis jarak, fitur dengan skala besar dapat mendominasi perhitungan.

\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Contoh}
\footnotesize
Misalkan dua fitur mempunyai rentang yang sangat berbeda:
\[
x_1\in[18,60]\quad\text{(usia)},
\]
\[
x_2\in[2000,20000]\quad\text{(pendapatan)}.
\]
Perbedaan pada pendapatan dapat jauh lebih besar secara numerik daripada perbedaan usia.
\end{block}
\column{.48\textwidth}
\begin{block}{Min-max normalization}
\footnotesize
Untuk setiap fitur,
\[
z
=
\frac{x-x_{\min}}
{x_{\max}-x_{\min}},
\]
sehingga nilai fitur berada pada rentang $[0,1]$.
\end{block}
\end{columns}

\begin{alertblock}{Prinsip}
\footnotesize
Gunakan normalisasi ketika skala fitur berbeda dan perbedaan skala tersebut bukan bagian dari makna yang ingin dipertahankan.
\end{alertblock}
\end{frame}

% 16
\begin{frame}[fragile]{Implementasi Fungsi Dasar dengan NumPy}
\begin{columns}[T]
\column{.52\textwidth}
\begin{lstlisting}
import numpy as np

def jarak_euclidean(x, mu):
    selisih = x - mu
    return np.sqrt(np.sum(selisih**2))

def centroid(X_cluster):
    return np.mean(X_cluster, axis=0)
\end{lstlisting}

\column{.44\textwidth}
\small
\begin{block}{Yang dihitung}
\footnotesize
Untuk $x,\mu\in\mathbb{R}^d$,
\[
\|x-\mu\|_2
=
\sqrt{\sum_{\ell=1}^{d}(x_\ell-\mu_\ell)^2}.
\]

Untuk satu cluster dengan $n_j$ anggota,
\[
X_{C_j}\in\mathbb{R}^{n_j\times d},
\qquad
\mu_j
=
\frac1{n_j}\sum_{i\in C_j}x_i.
\]
\end{block}
\end{columns}

\begin{alertblock}{Konsistensi dengan materi sebelumnya}
\footnotesize
Fungsi jarak sama dengan norma Euclidean yang telah digunakan pada KNN dan aljabar linear komputasional.
\end{alertblock}
\end{frame}

% 17
\begin{frame}[fragile]{Implementasi Tahap Assignment}
\begin{lstlisting}
def assignment(X, centroids):
    labels = []

    for x in X:
        distances = []

        for mu in centroids:
            d = jarak_euclidean(x, mu)
            distances.append(d)

        label = np.argmin(distances)
        labels.append(label)

    return np.array(labels)
\end{lstlisting}

\begin{block}{Interpretasi indeks}
\small
`np.argmin(distances)` mengembalikan posisi centroid dengan jarak terkecil. Jika hasilnya `0`, observasi ditempatkan pada cluster pertama.
\end{block}
\end{frame}

% 18
\begin{frame}[fragile]{Implementasi Tahap Update Centroid}
\begin{lstlisting}
def update_centroids(X, labels, k):
    centroids = []

    for j in range(k):
        Xj = X[labels == j]
        mu_j = np.mean(Xj, axis=0)
        centroids.append(mu_j)

    return np.array(centroids)
\end{lstlisting}

\begin{block}{Struktur data}
\small
Ekspresi \texttt{X[labels == j]} mengambil seluruh observasi yang saat ini berada pada cluster ke-$j$. Rata-rata kolom menghasilkan centroid baru di $\mathbb{R}^d$.
\end{block}
\end{frame}

% 19
\begin{frame}[fragile]{Implementasi Iterasi k-Means}
\begin{lstlisting}[basicstyle=\ttfamily\tiny]
def kmeans(X, k, centroids, max_iter=100, tol=1e-6):
    for it in range(max_iter):
        labels = assignment(X, centroids)
        new_centroids = update_centroids(X, labels, k)

        perubahan = np.linalg.norm(
            new_centroids - centroids
        )

        centroids = new_centroids

        if perubahan < tol:
            break

    return labels, centroids
\end{lstlisting}

\begin{block}{Kriteria berhenti}
\small
Algoritma berhenti ketika perubahan seluruh centroid cukup kecil,
\[
\|\mu^{\mathrm{baru}}-\mu^{\mathrm{lama}}\|_2<\texttt{tol},
\]
atau jumlah iterasi mencapai \texttt{max\_iter}.
\end{block}
\end{frame}

\begin{frame}[fragile]{Fungsi Objektif pada Implementasi}
\begin{columns}[T]
\column{.55\textwidth}
\begin{lstlisting}
def objective(X, labels, centroids):
    J = 0.0

    for i in range(len(X)):
        j = labels[i]
        diff = X[i] - centroids[j]
        J += np.dot(diff, diff)

    return J
\end{lstlisting}

\column{.41\textwidth}
\small
Kode tersebut menghitung
\[
J
=
\sum_{i=1}^{n}
\|x_i-\mu_{c_i}\|_2^2.
\]

\begin{block}{Penggunaan}
\footnotesize
Nilai $J$ berguna untuk:
\begin{itemize}
\item membandingkan beberapa inisialisasi;
\item memantau apakah iterasi membaik;
\item membantu menilai kandidat banyak cluster $k$.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% 21
\begin{frame}{Fungsi Kerja dan Kompleksitas Algoritma}
\small
Misalkan:
\[
n=\text{banyak observasi},\quad
d=\text{banyak fitur},\quad
k=\text{banyak cluster},\quad
m=\text{banyak iterasi}.
\]

Pada tahap assignment, setiap observasi dibandingkan dengan setiap centroid:
\[
n\times k
\]
perhitungan jarak, dan setiap jarak membaca $d$ komponen.

Maka satu iterasi mempunyai kerja utama
\[
T(n,k,d)\in O(nkd).
\]

Jika algoritma memerlukan $m$ iterasi,
\[
T(n,k,d,m)\in O(mnkd).
\]

\begin{block}{Hubungan dengan kode}
\footnotesize
Struktur loop `for x in X` dan `for mu in centroids` menjelaskan faktor $n$ dan $k$; operasi pada seluruh komponen vektor menjelaskan faktor $d$.
\end{block}
\end{frame}

% 22
\begin{frame}{Fungsi Objektif untuk Berbagai Banyak Cluster}
\footnotesize
Untuk setiap kandidat $k$, hasil k-Means dinilai dengan
\[
J(k)
=
\sum_{j=1}^{k}\sum_{i\in C_j}
\|x_i-\mu_j\|_2^2.
\]

Khusus $k=1$, seluruh observasi berada dalam satu cluster. Centroidnya adalah
\[
\bar x=\frac1n\sum_{i=1}^{n}x_i,
\]
sehingga
\[
\boxed{
J(1)=\sum_{i=1}^{n}\|x_i-\bar x\|_2^2
}.
\]

Untuk solusi optimal pada setiap $k$,
\[
J(k+1)\le J(k).
\]

{\scriptsize Jadi penambahan cluster selalu dapat mempertahankan atau menurunkan within-cluster sum of squares. Elbow method digunakan untuk menilai kapan penurunan tersebut mulai tidak terlalu besar.}
\sumber{James et al. (2021), \S12.4; Hastie et al. (2009), \S14.3.}
\end{frame}

\end{alertblock}

\begin{frame}{Batasan dan Pemeriksaan Hasil k-Means}
\small
\begin{itemize}
\item Hasil bergantung pada \textbf{skala fitur}; normalisasi sering diperlukan.
\item Hasil dapat bergantung pada \textbf{centroid awal}; lakukan beberapa inisialisasi.
\item k-Means menggunakan rata-rata, sehingga \textbf{outlier} dapat menarik centroid.
\item Pembagian berdasarkan centroid cenderung cocok untuk kelompok yang relatif kompak dalam metrik Euclidean.
\item Nilai $k$ harus dipilih; tidak selalu ada satu pilihan yang jelas.
\end{itemize}
