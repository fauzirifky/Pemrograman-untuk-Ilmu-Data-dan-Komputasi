\begin{frame}{Halt atau terminasi}
\begin{block}{Definisi kerja}
Suatu program dikatakan \textit{halt} untuk sebuah input jika program berhenti setelah jumlah langkah yang berhingga.
\end{block}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Loop}
Kita mencari ukuran yang bergerak menuju kondisi berhenti.
\[
 i\to i-1,\quad i\to i//2,\quad |x_k^2-n|\to 0.
\]
\end{block}
\column{.48\textwidth}
\begin{block}{Rekursi}
Kita mencari submasalah yang lebih kecil dan base case.
\[
 n! = n(n-1)!,\quad 0! = 1.
\]
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Mengapa bukan except kosong?}
\begin{block}{Masalah dengan \texttt{except:}}
\texttt{except:} menangkap terlalu banyak hal. Bug yang seharusnya terlihat bisa tersembunyi.
\end{block}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Lebih baik}
Tangani jenis masalah yang kita pahami:
\begin{itemize}\small
  \item \texttt{ValueError}: nilai tidak dapat dikonversi.
  \item \texttt{TypeError}: tipe data tidak sesuai.
  \item \texttt{ZeroDivisionError}: pembagian nol.
\end{itemize}
\end{block}
\column{.48\textwidth}
\begin{block}{Prinsip}
Jika error belum dipahami, biarkan terlihat saat pengembangan. Setelah jelas, baru tangani.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Masalah 3: akar kuadrat dengan Newton}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Masalah matematika}
Cari \(x\) sehingga
\[
 x^2=n.
\]
Definisikan \(F(x)=x^2-n\). Akar \(F(x)=0\) adalah \(\sqrt n\).
\end{block}
\column{.48\textwidth}
\begin{block}{Iterasi Newton}
\[
 x_{k+1}=x_k-\frac{F(x_k)}{F'(x_k)}
 =\frac12\left(x_k+\frac{n}{x_k}\right).
\]
Syarat penting: \(x_k\neq0\).
\end{block}
\end{columns}
\end{frame}

% 14
\begin{frame}{Hitung manual Newton untuk \(\sqrt{10}\)}
Ambil \(x_0=5\). Rumus:
\[
 x_{k+1}=\frac12\left(x_k+\frac{10}{x_k}\right).
\]
\begin{center}
\begin{tabular}{c|c|c}
\toprule
\(k\) & \(x_k\) & \(x_k^2-10\)\\
\midrule
0 & 5.0000 & 15.0000\\
1 & 3.5000 & 2.2500\\
2 & 3.1786 & 0.1033\\
3 & 3.1623 & mendekati 0\\
\bottomrule
\end{tabular}
\end{center}
\begin{block}{Yang diamati}
Galat \(|x_k^2-n|\) semakin kecil. Inilah alasan kita boleh berhenti setelah toleransi tercapai.
\end{block}
\end{frame}

\begin{frame}[fragile]{Newton yang lebih aman}
\begin{lstlisting}
def akar(n, tol=1e-10, max_iter=100):
    if n < 0:
        return None
    if n == 0:
        return 0.0
    x = n / 2 if n >= 1 else 1.0
    for k in range(max_iter):
        if abs(x*x - n) <= tol:
            return x
        x = 0.5 * (x + n/x)
    return None
\end{lstlisting}
\begin{block}{Makna desain}
Ada penanganan domain, tebakan awal tidak nol, toleransi, dan batas iterasi.
\end{block}

\begin{frame}{Masalah 4: fungsi prima dan kondisi batas}
\begin{block}{Definisi}
Bilangan bulat positif \(n\) disebut prima jika \(n>1\) dan hanya punya pembagi positif 1 dan \(n\).
\end{block}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Mengapa cukup sampai \(\sqrt n\)?}
Jika \(n=ab\) dan \(a>\sqrt n\), maka \(b<\sqrt n\). Jadi salah satu faktor nontrivial pasti tidak melebihi \(\sqrt n\).
\end{block}
\column{.48\textwidth}
\begin{block}{Contoh manual}
Untuk \(n=49\), \(\sqrt{49}=7\). Jika percobaan pembagi berhenti sebelum 7, faktor 7 tidak pernah diuji.
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Kode prima yang salah}
\begin{columns}[T]
\column{.56\textwidth}
\begin{lstlisting}
def is_prima(n):
    if n <= 1:
        return False
    d = 2
    while d*d < n:   # salah di batas
        if n % d == 0:
            return False
        d += 1
    return True
\end{lstlisting}
\column{.40\textwidth}
\begin{block}{Cari pembongkar bug}
Uji:
\[
 n=4,\ 9,\ 25,\ 49.
\]
Apa yang terjadi ketika \(d^2=n\)?
\end{block}
\end{columns}
\vspace{1mm}
\colorbox{LightTeal}{\parbox{.96\linewidth}{\small Perbaikan: gunakan \texttt{while d*d <= n}. Kondisi batas sering lebih penting daripada kasus besar.}}
\end{frame}

\begin{frame}{Jarak Euclid: dari rumus ke algoritma}
Untuk dua vektor
\[
 x=(x_1,\ldots,x_m),\qquad y=(y_1,\ldots,y_m),
\]
jarak Euclid didefinisikan sebagai
\[
 d(x,y)=\sqrt{\sum_{i=1}^{m}(x_i-y_i)^2}.
\]
\begin{block}{Algoritma manual}
\begin{enumerate}\small
  \item Pastikan panjang \(x\) dan \(y\) sama.
  \item Untuk tiap indeks, hitung selisih \(x_i-y_i\).
  \item Kuadratkan selisih, jumlahkan, lalu ambil akar.
\end{enumerate}
\end{block}
\end{frame}

from math import sqrt

def jarak(x, y):
    assert len(x) == len(y)

\end{lstlisting}

assert jarak([0,0],[3,4]) == 5

\end{lstlisting}
\end{block}

\begin{frame}{Rekursi dari masalah matematika}
\begin{block}{Pertanyaan matematika}
Bagaimana menghitung \(5!\)?
\[
5! = 5\cdot4\cdot3\cdot2\cdot1.
\]
\end{block}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Pola yang terlihat}
\[
5! = 5\cdot4!,\quad
4! = 4\cdot3!,\quad
3! = 3\cdot2!.
\]
\end{block}
\column{.48\textwidth}
\begin{block}{Definisi rekursif}
\[
 n! = n(n-1)!,\quad 0! = 1.
\]
Base case: \(0!\). Recursive case: \(n(n-1)!\).
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Faktorial: rekursif dan iteratif}
\begin{columns}[T]
\column{.48\textwidth}
\begin{lstlisting}
def faktorial(n):
    if n == 0:
        return 1
    return n * faktorial(n-1)
\end{lstlisting}
\column{.48\textwidth}
\begin{lstlisting}
def faktorial_iter(n):
    hasil = 1
    for k in range(1, n+1):
        hasil *= k
    return hasil
\end{lstlisting}
\end{columns}
\begin{block}{Diskusi}
Keduanya menghitung nilai yang sama. Versi rekursif menyimpan pemanggilan yang belum selesai; versi iteratif menyimpan akumulasi \texttt{hasil}.
\end{block}
\end{frame}

\begin{frame}[fragile]{Kode rekursif barisan}

def a(n):

        return 2

assert a(0) == 2

\end{columns}
\vspace{1mm}
\begin{block}{Rumus eksplisit}
Dengan mengamati nilai: \(2,4,10,28,\ldots\), mahasiswa dapat menebak pola, lalu memverifikasi dengan induksi atau pengujian numerik.

% 27
\begin{frame}{Mencari rumus eksplisit secara bertahap}
Untuk
\[
 a_n = 3a_{n-1}-2,
\]
carilah titik tetap \(L
i=3L-2\), sehingga
\[
 L=1.
\]
Definisikan \(b_n=a_n-1\). Maka
\[
 b_n = a_n-1 = 3a_{n-1}-3 = 3(a_{n-1}-1)=3b_{n-1}.
\]
Karena \(b_0=a_0-1=1\), maka
\[
 b_n=3^n,
 \qquad a_n=3^n+1.
\]
\end{frame}

\begin{frame}[fragile]{Kode rekursif langsung: benar, tetapi bisa mahal}
\begin{columns}[T]
\column{.52\textwidth}
\begin{lstlisting}
def u(n):
    if n == 0:
        return 1
    if n == 1:
        return 3
    return u(n-1) + 2*u(n-2)
\end{lstlisting}
\column{.44\textwidth}
\begin{block}{Untuk \(u_5\)}
\[
\begin{aligned}
u_2&=3+2(1)=5,\\
u_3&=5+2(3)=11,\\
u_4&=11+2(5)=21,\\
u_5&=21+2(11)=43.
\end{aligned}
\]
\end{block}
\end{columns}
\begin{block}{Diskusi}
Kalau nilai kecil dapat dihitung manual dengan tabel, mengapa rekursi langsung tetap melakukan banyak pemanggilan ulang?
\end{block}
\end{frame}

\begin{frame}{Dari recursive case ke tabel iteratif}
\begin{columns}[T]
\column{.46\textwidth}
\begin{block}{Definisi}
\[
 u_n=u_{n-1}+2u_{n-2}.
\]
Untuk menghitung \(u_n\), sebenarnya cukup menyimpan dua nilai terakhir.
\end{block}
\column{.50\textwidth}
\begin{center}
\begin{tabular}{c|cc|c}
\toprule
n & \(u_{n-2}\) & \(u_{n-1}\) & \(u_n\)\\
\midrule
2 & 1 & 3 & 5\\
3 & 3 & 5 & 11\\
4 & 5 & 11 & 21\\
5 & 11 & 21 & 43\\
\bottomrule
\end{tabular}
\end{center}
\end{columns}
\begin{block}{Pesan}
Rekursi membantu mendeskripsikan masalah. Implementasi tidak selalu harus rekursif jika tabel iteratif lebih jelas.
\end{block}
\end{frame}

\begin{frame}{Membaca kode salah: prosedur kelas}
\begin{enumerate}
  \item Tulis satu input kecil yang jawabannya dapat dihitung manual.
  \item Buat tabel nilai variabel per langkah.
  \item Tandai baris pertama yang membuat nilai keluar dari harapan.
  \item Perbaiki satu hal saja.
  \item Uji ulang dengan kasus normal dan kasus batas.
\end{enumerate}
\begin{block}{Aturan diskusi}
Kalau maju ke papan, tidak harus langsung benar. Yang dicari adalah alasan mengapa program salah.
\end{block}
\end{frame}

\begin{frame}[fragile]{Latihan kelas: diagnosa cepat}
\begin{lstlisting}
def jumlah_positif(data):
    total = 0
    for x in data:
        if x > 0:
            total += x
        else:
            return total
    return total
\end{lstlisting}
\begin{block}{Pertanyaan}
Untuk \texttt{data = [2, 3, -1, 4, 5]}, apakah fungsi memang menjumlahkan semua bilangan positif? Jika tidak, baris mana yang menyebabkan masalah?
\end{block}
\end{frame}
