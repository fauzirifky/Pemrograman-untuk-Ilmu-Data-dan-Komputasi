% 13
\begin{frame}{Masalah matematis: menghitung akar kuadrat}
\begin{columns}[T]
\column{.46\textwidth}
\begin{block}{Tujuan}
Mencari pendekatan untuk $\sqrt{n}$ berarti mencari $x$ sehingga
\[
 x^2-n=0.
\]
\end{block}
\begin{block}{Newton-Raphson}
\[
 x_{k+1}=x_k-\frac{f(x_k)}{f'(x_k)}.
\]
\end{block}
\column{.50\textwidth}
\begin{block}{Untuk $f(x)=x^2-n$}
\[
 f'(x)=2x
\]
\[
 x_{k+1}=\frac12\left(x_k+\frac{n}{x_k}\right).
\]
\end{block}
\begin{itemize}\small
  \item Ini contoh loop yang harus memiliki alasan berhenti.
  \item Perbaikan berhenti dapat memakai toleransi atau batas iterasi.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Versi yang dapat diperiksa}
\begin{lstlisting}
def akar_newton(n, tol=1e-8, maks_iterasi=20):
    if n < 0:
        return None
    if n == 0:
        return 0.0
    x = n / 2
    for k in range(maks_iterasi):
        x_baru = 0.5 * (x + n/x)
        if abs(x_baru - x) < tol:
            return x_baru
        x = x_baru
    return x
\end{lstlisting}

  \item Ada tebakan awal.
  \item Ada kondisi berhenti dan batas iterasi.
\end{itemize}

% 17
\begin{frame}[fragile]{Implementasi rekursi: faktorial}
\begin{columns}[T]
\column{.56\textwidth}
\begin{lstlisting}
def faktorial(n):
    if n == 0:
        return 1
    return n * faktorial(n - 1)

print(faktorial(5))
\end{lstlisting}
\column{.40\textwidth}
\begin{block}{Dua bagian}
\begin{itemize}
  \item Base case: \texttt{n == 0}.
  \item Recursive case: \texttt{faktorial(n-1)}.
\end{itemize}
\end{block}
\begin{block}{Telusuri}
$5! \rightarrow 4! \rightarrow 3! \rightarrow 2! \rightarrow 1! \rightarrow 0!$
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Rekursi untuk barisan}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Masalah}
Diberikan barisan
\[
 a_0=2, \qquad a_n=3a_{n-1}+1.
\]
Hitung $a_1$, $a_2$, dan $a_3$.
\end{block}
\column{.48\textwidth}
\begin{block}{Struktur rekursif}
\[
\begin{aligned}
a_1 &= 3a_0+1,\\
a_2 &= 3a_1+1,\\
a_3 &= 3a_2+1.
\end{aligned}
\]
Setiap langkah membutuhkan nilai sebelumnya.
\end{block}
\end{columns}
\end{frame}

def fib(n):
