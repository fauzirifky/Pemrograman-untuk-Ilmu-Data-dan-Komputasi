def rasio(a, b):

\begin{frame}[fragile]{Masalah 3: Newton untuk akar kuadrat}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Dari matematika}
Untuk mencari $\sqrt{n}$, gunakan iterasi
\[
x_{k+1}=\frac12\left(x_k+\frac{n}{x_k}\right).
\]
\end{block}
\begin{block}{Hitung manual dulu}
Untuk $n=9$ dan $x_0=4.5$, hitung $x_1,x_2,x_3$.
\end{block}
\column{.48\textwidth}
\only<1>{\begin{alertblock}{Tahan dulu}\centering Jangan buka kalkulator dulu. Kerjakan satu iterasi bersama.\end{alertblock}}
\only<2->{%
\[
\begin{aligned}
x_1 &= \tfrac12(4.5+9/4.5)=3.25,\\
x_2 &\approx 3.0096,\\
x_3 &\approx 3.0000.
\end{aligned}
\]
}
\end{columns}
\end{frame}

% 8
\begin{frame}[fragile]{Kode Newton yang salah: apa masalahnya?}
\begin{columns}[T]
\column{.55\textwidth}
\begin{lstlisting}
def akar(n, tol=1e-10):
    x = 0.0
    while abs(x*x - n) > tol:
        x = 0.5 * (x + n/x)
    return x
\end{lstlisting}
\column{.40\textwidth}
\begin{block}{Tebak dulu}
\begin{enumerate}\small
  \item Apa yang terjadi saat $n=9$?
  \item Apa yang terjadi saat $n=0$?
  \item Bagaimana jika $n<0$?
\end{enumerate}
\end{block}
\pause
\begin{alertblock}{Masalah utama}
Tebakan awal $x=0$ membuat $n/x$ gagal. Kasus $n=0$ dan $n<0$ juga harus dipikirkan.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Algoritma Newton yang lebih aman}
\begin{lstlisting}
def akar(n, tol=1e-10, max_iter=50):
    if n < 0:
        return None
    if n == 0:
        return 0.0

    x = n / 2 if n >= 1 else 1.0
    for k in range(max_iter):
        x_baru = 0.5 * (x + n/x)
        if abs(x_baru*x_baru - n) < tol:
            return x_baru
        x = x_baru
    return x
\end{lstlisting}
\pause
\begin{block}{Uji minimal}
Coba $n=0,1,4,9,2,-1$. Bandingkan dengan nilai yang sudah diketahui.
\end{block}
\end{frame}

\begin{frame}[fragile]{Faktorial: rekursif dan iteratif}
\begin{columns}[T]
\column{.48\textwidth}
\begin{lstlisting}
def faktorial(n):
    if n == 0:
        return 1
    return n * faktorial(n-1)
\end{lstlisting}
\column{.48\textwidth}
\begin{lstlisting}
def faktorial_iter(n):
    hasil = 1
    for k in range(1, n+1):
        hasil *= k
    return hasil
\end{lstlisting}
\end{columns}

\begin{block}{Telusuri call stack}

\end{block}

\begin{frame}[fragile]{Testing dengan assert}
\begin{lstlisting}
import math

def jarak(x, y):
    assert len(x) == len(y)
    s = 0
    for i in range(len(x)):
        s += (x[i] - y[i])**2
    return math.sqrt(s)

assert jarak([0,0], [0,0]) == 0
assert jarak([0,0], [3,4]) == 5
assert jarak([3,4], [0,0]) == jarak([0,0], [3,4])
\end{lstlisting}

\end{block}

\begin{frame}[fragile]{Masalah 9: bilangan prima dan kondisi batas}
\begin{columns}[T]
\column{.55\textwidth}
\begin{lstlisting}
def prima(n):
    if n <= 1:
        return False
    pembagi = 2
    while pembagi*pembagi < n:
        if n % pembagi == 0:
            return False
        pembagi += 1
    return True
\end{lstlisting}
\column{.40\textwidth}
\begin{block}{Prediksi}
Uji secara mental untuk:
\begin{itemize}\small
  \item $n=2$
  \item $n=9$
  \item $n=49$
\end{itemize}
Mana yang membongkar bug?
\end{block}
\pause
\begin{alertblock}{Masalahnya}
Syarat \texttt{< n} melewatkan pembagi saat \texttt{pembagi*pembagi == n}. Gunakan \texttt{<= n}.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Langkah membaca kode bermasalah}
\begin{enumerate}
  \item Ambil contoh kecil yang bisa dihitung manual.
  \item Catat nilai variabel setelah setiap baris penting.
  \item Tanya: apakah loop atau rekursi mendekati kondisi berhenti?
  \item Buat satu test yang seharusnya benar.
  \item Buat satu test batas: nol, data kosong, bilangan negatif, panjang berbeda.
  \item Perbaiki kode, lalu jalankan ulang test.
\end{enumerate}
\end{frame}
