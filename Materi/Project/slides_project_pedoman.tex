% 7
\begin{frame}{Kriteria minimum program}
\small
\begin{enumerate}
  \item Program menghasilkan solusi komputasi/sains data yang sesuai dengan masalah.
  \item Memuat minimal \textbf{3 fungsi buatan sendiri} yang memiliki peran nyata.
  \item Percabangan dan/atau perulangan dipakai secara \textbf{bermakna}, bukan sekadar ditambahkan agar ``ada loop''.
  \item Menggunakan struktur data yang sesuai: list, array/NumPy, atau dictionary.
  \item Fungsi utama terdokumentasi: \textbf{kegunaan, parameter/input, tipe data, dan output/\texttt{return}}.
  \item Kode tersusun modular; library digunakan untuk mendukung solusi, bukan menggantikan seluruh proses berpikir.
\end{enumerate}
\vspace{2mm}
\begin{alertblock}{Beyond ``programnya jalan''}
Dosen dapat mengubah input, meminta tracing, menanyakan tipe data, batas input, kemungkinan error, dan alasan pemilihan struktur/algoritma.
\end{alertblock}
\end{frame}

\begin{frame}{Bagaimana code defense dilakukan?}
\begin{columns}[T]
\column{.49\textwidth}
\begin{block}{Pemilihan mahasiswa}
\small
Dosen dapat memilih anggota secara \textbf{acak/representatif}, termasuk anggota yang tampak paling memahami dan paling kurang memahami program.
\end{block}
\begin{block}{Yang dapat diuji}
\small
\begin{itemize}
  \item ``Jelaskan bagian kode ini.''
  \item ``Apa tipe data variabel ini?''
  \item ``Jika input saya ubah, apa yang terjadi?''
  \item ``Input apa yang membuat fungsi ini gagal/tidak sesuai?''
  \item ``Mengapa loop/if ini diperlukan?''
\end{itemize}
\end{block}
\column{.47\textwidth}
\begin{block}{Konsekuensi penilaian}
\small
\begin{itemize}
  \item Nilai code defense adalah \textbf{individual}.
  \item Anggota satu kelompok dapat memperoleh nilai akhir project yang berbeda.
  \item Program yang benar tetapi tidak dipahami tidak memperoleh penilaian penuh.
\end{itemize}
\end{block}
\begin{alertblock}{Inti}
\small
Yang diuji bukan siapa yang paling cepat mengetik kode, tetapi siapa yang \textbf{memahami proses komputasi} yang dikumpulkan kelompoknya.
\end{alertblock}
\end{columns}
\end{frame}

% 10
\begin{frame}{Menggunakan AI dengan bijak}
\footnotesize
\begin{columns}[T]
\column{.49\textwidth}
\begin{block}{AI boleh digunakan untuk}
\begin{itemize}
  \item berdiskusi dan menggali alternatif ide;
  \item belajar konsep atau cara kerja algoritma;
  \item meminta contoh sederhana sebelum membuat solusi sendiri;
  \item memperoleh petunjuk debugging dan melakukan code review;
  \item memperbaiki bahasa/struktur tulisan tanpa mengubah substansi ilmiah.
\end{itemize}
\end{block}
\column{.49\textwidth}
\begin{block}{Deklarasikan penggunaan AI}
Sebutkan minimal:
\begin{enumerate}
  \item alat/model AI yang digunakan;
  \item bagian project yang dibantu;
  \item bentuk bantuan: ide, kode, debugging, analisis, atau penulisan;
  \item cara tim memeriksa kembali hasil AI;
  \item bagian penting yang akhirnya dipahami/dikerjakan sendiri oleh tim.
\end{enumerate}
\end{block}
\end{columns}
\begin{alertblock}{Tidak dapat didelegasikan ke AI}
\footnotesize Pemahaman kode, validasi data/hasil, ketepatan sumber, dan tanggung jawab akademik tetap berada pada mahasiswa.
\end{alertblock}
\vspace{-1mm}\scriptsize\textit{Prinsip: penggunaan GenAI harus transparan dan hasilnya diverifikasi oleh manusia.}
\end{frame}
