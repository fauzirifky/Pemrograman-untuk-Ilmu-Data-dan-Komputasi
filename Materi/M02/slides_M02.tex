% 3
\begin{frame}{Apa itu subprogram?}
\begin{columns}[T]
\column{.52\textwidth}
\begin{block}{Makna umum}
Subprogram adalah bagian program yang diberi nama untuk menjalankan satu tugas tertentu. Subprogram dapat dipanggil dari bagian lain program.
\end{block}
\begin{block}{Dalam Python}
Bentuk utama subprogram adalah fungsi yang ditulis dengan \texttt{def}. Fungsi dapat menerima input melalui parameter dan mengirim hasil melalui \texttt{return}.
\end{block}
\column{.44\textwidth}
\begin{center}
\begin{tikzpicture}[node distance=8mm]
\node[box,minimum width=3.8cm] (main) {program utama};
\node[darkbox,minimum width=3.8cm,below=of main] (sub) {subprogram};
\node[box,minimum width=3.8cm,below=of sub] (out) {hasil kembali};
\draw[arr] (main) -- node[right]{panggil} (sub);
\draw[arr] (sub) -- node[right]{return} (out);
\end{tikzpicture}
\end{center}
\end{columns}
\vspace{1mm}
\colorbox{LightTeal}{\parbox{.96\linewidth}{\small Subprogram membuat program lebih mudah dibaca, diuji, dan digunakan ulang.}}
\end{frame}

\begin{frame}{Fungsi sebagai aturan transformasi}
\begin{columns}[T]
\column{.45\textwidth}
\vspace{6mm}
\[
\boxed{y=f(x)}
\]
\[
\boxed{\text{input} \rightarrow \text{proses} \rightarrow \text{output}}
\]
\begin{block}{Dalam matematika}
Fungsi menghubungkan masukan dengan keluaran menurut aturan tertentu.
\end{block}
\column{.51\textwidth}
\begin{center}
\begin{tikzpicture}[node distance=7mm]
\node[box,minimum width=2.2cm] (x) {$x$};
\node[darkbox,minimum width=2.6cm,right=of x] (f) {$f$};
\node[box,minimum width=2.5cm,right=of f] (y) {$f(x)$};
\draw[arr] (x) -- (f);
\draw[arr] (f) -- (y);
\node[below=4mm of f,align=center,text width=6.2cm] {Dalam program, fungsi menyimpan aturan itu dalam bentuk langkah-langkah yang dapat dijalankan komputer.};
\end{tikzpicture}
\end{center}
\begin{block}{Dalam Python}
\texttt{def} memberi nama pada aturan komputasi. Parameter menjadi masukan. \texttt{return} menjadi keluaran.
\end{block}
\end{columns}
\end{frame}

% 5
\begin{frame}[fragile]{Anatomi fungsi Python}
\begin{lstlisting}
def nama_fungsi(parameter_1, parameter_2):
    langkah_1
    langkah_2
    return hasil
\end{lstlisting}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Bagian definisi}
\begin{itemize}
  \item \texttt{def}: mendefinisikan fungsi.
  \item Nama fungsi: tugas yang dikerjakan.
  \item Parameter: nama input di dalam fungsi.
\end{itemize}
\end{block}
\column{.48\textwidth}
\begin{block}{Saat dipanggil}
\begin{itemize}
  \item Argumen: nilai yang dikirim ke fungsi.
  \item \texttt{return}: nilai yang dikirim kembali.
  \item Setelah \texttt{return}, fungsi selesai.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

def luas_persegi_panjang(panjang, lebar):

\begin{frame}[fragile]{Fungsi dengan argumen}
\begin{columns}[T]

def tampil_kuadrat(x):

% 10
\begin{frame}[fragile]{\texttt{print} tidak sama dengan \texttt{return}}
\begin{columns}[T]
\column{.48\textwidth}
\begin{lstlisting}
def g(x):
    return x**2 - 1

def h(x):
    print(g(x))

nilai = h(3)
print(nilai)
\end{lstlisting}
\column{.48\textwidth}
\begin{lstlisting}
z = h(3) + 2
# TypeError
\end{lstlisting}
\begin{block}{Ide utama}
\texttt{print} hanya menampilkan nilai. Jika tidak ada \texttt{return}, fungsi mengembalikan \texttt{None}.
\end{block}
\end{columns}
\end{frame}

% 11
\begin{frame}[fragile]{Fungsi yang mengembalikan lebih dari satu nilai}
\begin{lstlisting}
def stat_ringkas(a, b, c, d):
    nilai = [a, b, c, d]
    minimum = min(nilai)
    maksimum = max(nilai)
    rata = sum(nilai) / len(nilai)
    rentang = maksimum - minimum
    return minimum, maksimum, rata, rentang

mn, mx, r, rg = stat_ringkas(4, 7, 1, 9)
\end{lstlisting}
\begin{block}{Yang terjadi}
Python sebenarnya mengembalikan satu objek bertipe tuple. Unpacking membagi isi tuple ke beberapa variabel.
\end{block}
\end{frame}

\begin{frame}[fragile]{Tuple dan unpacking}
\begin{columns}[T]
\column{.48\textwidth}
\begin{lstlisting}
hasil = stat_ringkas(4, 7, 1, 9)

print(type(hasil))

minimum, maksimum, rata, rentang = hasil
\end{lstlisting}
\column{.48\textwidth}
\begin{lstlisting}
# jumlah variabel harus cocok
x, y = hasil
# ValueError
\end{lstlisting}
\begin{block}{Mengapa penting?}
Satu proses sering menghasilkan beberapa informasi: nilai estimasi, galat, jumlah iterasi, atau status konvergensi.
\end{block}
\end{columns}
\end{frame}

% 13
\begin{frame}[fragile]{Scope: variabel lokal dan global}
\begin{lstlisting}
x = 10

def f(a):
    x = a + 2
    return x

print(f(5))
print(x)
\end{lstlisting}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Variabel lokal}
\texttt{x} di dalam fungsi hanya hidup di dalam fungsi tersebut.
\end{block}
\column{.48\textwidth}
\begin{block}{Variabel global}
\texttt{x} di luar fungsi tetap bernilai 10. Nilai ini tidak otomatis berubah hanya karena ada \texttt{x} lokal.
\end{block}
\end{columns}
\end{frame}

% 14
\begin{frame}[fragile]{Mengapa variabel global perlu hati-hati?}
\begin{lstlisting}
total = 0

def tambah(x):
    global total
    total = total + x
    return total
\end{lstlisting}
\begin{block}{Masalah yang sering muncul}
\begin{itemize}
  \item Hasil fungsi bergantung pada riwayat pemanggilan.
  \item Test menjadi tidak mandiri.
  \item Kesalahan sulit dilacak karena nilai dapat berubah dari tempat lain.
\end{itemize}
\end{block}
\colorbox{LightTeal}{\parbox{.96\linewidth}{\small Utamakan fungsi yang menerima input secara eksplisit dan mengembalikan output secara eksplisit.}}
\end{frame}

% 15
\begin{frame}[fragile]{Fungsi sebagai alat dekomposisi}
\begin{columns}[T]
\column{.43\textwidth}
\[
d(p,q)=\sqrt{(p_1-q_1)^2+(p_2-q_2)^2}
\]
\[
E(p,q)=\frac{1}{1+d(p,q)^2}
\]
\column{.53\textwidth}
\begin{lstlisting}
def jarak(p, q):
    dx = p[0] - q[0]
    dy = p[1] - q[1]
    return (dx**2 + dy**2)**0.5

def energi(d):
    return 1 / (1 + d**2)

def energi_dua_titik(p, q):
    return energi(jarak(p, q))
\end{lstlisting}
\end{columns}
\end{frame}

% 16
\begin{frame}{Komposisi fungsi}
\vspace{1mm}
Jika $g$ menerima $x$, dan $f$ menerima hasil dari $g$, maka
\[
(f\circ g)(x)=f(g(x)).
\]
\vspace{2mm}
\begin{center}
\begin{tikzpicture}[node distance=8mm]
\node[box,minimum width=1.5cm] (x) {$x$};
\node[darkbox,minimum width=1.8cm,right=of x] (g) {$g$};
\node[box,minimum width=2.1cm,right=of g] (gx) {$g(x)$};
\node[darkbox,minimum width=1.8cm,right=of gx] (f) {$f$};
\node[box,minimum width=2.25cm,right=of f] (fg) {$f(g(x))$};
\draw[arr] (x) -- (g);
\draw[arr] (g) -- (gx);
\draw[arr] (gx) -- (f);
\draw[arr] (f) -- (fg);
\end{tikzpicture}
\end{center}
\vspace{2mm}
\begin{block}{Dalam Python}
Ide yang sama muncul saat satu fungsi memanggil fungsi lain. Program menjadi susunan unit kecil, bukan satu blok panjang.
\end{block}
\end{frame}

% 17
\begin{frame}[fragile]{Fungsi sebagai objek}
\begin{lstlisting}
def proses(x, f):
    return f(x)

def kuadrat(x):
    return x**2

def tambah1(x):
    return x + 1

print(proses(5, kuadrat))
print(proses(5, tambah1))
\end{lstlisting}
\begin{block}{Makna penting}
Nama \texttt{kuadrat} tanpa tanda kurung adalah objek fungsi. Objek itu dapat dikirim sebagai argumen ke fungsi lain.
\end{block}
\end{frame}

% 18
\begin{frame}[fragile]{List fungsi dan pipeline}
\begin{lstlisting}
def kuadrat(x):
    return x**2

def tambah1(x):
    return x + 1

def pipeline(x, ops):
    hasil = x
    for f in ops:
        hasil = f(hasil)
    return hasil

ops = [abs, kuadrat, tambah1]
print(pipeline(-3, ops))
\end{lstlisting}
\begin{block}{Baca alurnya}
Nilai $-3$ diproses oleh \texttt{abs}, lalu \texttt{kuadrat}, lalu \texttt{tambah1}. Pola ini sering muncul pada transformasi data.
\end{block}
\end{frame}

% 19
\begin{frame}[fragile]{Lambda: fungsi pendek untuk operasi pendek}
\begin{columns}[T]
\column{.42\textwidth}
\[
f(x)=3x^2-2x+1
\]
\begin{lstlisting}
f = lambda x: 3*x**2 - 2*x + 1
\end{lstlisting}
\column{.54\textwidth}
\begin{lstlisting}
xnilai = range(-3, 4)

y = [f(x) for x in xnilai]

z = list(map(lambda x: 3*x**2 - 2*x + 1,
             xnilai))
\end{lstlisting}
\end{columns}
\begin{block}{Gunakan secara wajar}
Lambda cocok untuk fungsi satu baris yang jelas. Jika logika mulai panjang, fungsi bernama lebih baik.
\end{block}
\end{frame}

\begin{frame}[fragile]{List comprehension untuk evaluasi fungsi}
\begin{lstlisting}
def f(x):

xnilai = [-3, -2, -1, 0, 1, 2, 3]
y = [f(x) for x in xnilai]

\begin{block}{Cara membaca}
Untuk setiap \texttt{x} di \texttt{xnilai}, hitung \texttt{f(x)}, lalu simpan hasilnya ke list baru.
\end{block}
\colorbox{LightTeal}{\parbox{.96\linewidth}{\small List dipakai sebagai wadah data sederhana. Struktur data dan array NumPy dibahas lebih formal pada pertemuan berikutnya.}}
\end{frame}

% 21
\begin{frame}{Rekursi: fungsi yang memanggil dirinya sendiri}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Ide utama}
Masalah dipecah menjadi versi yang lebih kecil dari masalah yang sama.
\end{block}
\begin{block}{Dua bagian wajib}
\begin{itemize}
  \item \textbf{Base case}: kondisi berhenti.
  \item \textbf{Recursive case}: pemanggilan fungsi pada masalah yang lebih kecil.
\end{itemize}
\end{block}
\column{.48\textwidth}
\begin{center}
\begin{tikzpicture}[node distance=6mm]
\node[darkbox,minimum width=3.1cm] (n) {masalah $n$};
\node[box,minimum width=3.1cm,below=of n] (n1) {masalah $n-1$};
\node[box,minimum width=3.1cm,below=of n1] (base) {base case};
\draw[arr] (n) -- (n1);
\draw[arr] (n1) -- (base);
\draw[arr] ([xshift=-7mm]base.west) .. controls +(-1.1,0) and +(-1.1,0) .. node[left]{kembali} ([xshift=-7mm]n.west);
\end{tikzpicture}
\end{center}
\end{columns}
\end{frame}

% 22
\begin{frame}[fragile]{Contoh rekursi: faktorial}
\begin{columns}[T]
\column{.42\textwidth}
\[
n! = n(n-1)!, \qquad 0! = 1.
\]
\[
5! = 5\cdot 4\cdot 3\cdot 2\cdot 1.
\]
\column{.54\textwidth}
\begin{lstlisting}
def faktorial(n):
    if n == 0:
        return 1
    return n * faktorial(n - 1)

print(faktorial(5))
\end{lstlisting}
\end{columns}
\begin{block}{Telusuri}
Tuliskan urutan pemanggilan \texttt{faktorial(5)}, \texttt{faktorial(4)}, \ldots, sampai kembali menghasilkan nilai akhir.
\end{block}
\end{frame}

% 23
\begin{frame}[fragile]{Rekursi untuk barisan}
Diberikan barisan
\[
a_0=2,\qquad a_{n}=3a_{n-1}+1, \quad n\ge 1.
\]
\begin{lstlisting}
def a(n):
    if n == 0:
        return 2
    return 3*a(n - 1) + 1
\end{lstlisting}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Tugas kecil}
Hitung manual $a_1$, $a_2$, dan $a_3$.
\end{block}
\column{.48\textwidth}
\begin{block}{Yang perlu diamati}
Setiap pemanggilan bergerak dari $n$ ke $n-1$, sehingga mendekati base case.
\end{block}
\end{columns}
\end{frame}

% 24
\begin{frame}[fragile]{Rekursi dan terminasi}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Berhenti}
\begin{lstlisting}
def turun(n):
    if n == 0:
        return 0
    return turun(n - 1)
\end{lstlisting}
\end{block}
\column{.48\textwidth}
\begin{block}{Tidak bergerak ke kondisi berhenti}
\begin{lstlisting}
def salah(n):
    if n == 0:
        return 0
    return salah(n + 1)

Saat melihat fungsi baru, tanyakan hal berikut.
\begin{enumerate}
  \item Apa tugas yang dikerjakan fungsi?
  \item Apa input yang dibutuhkan?
  \item Apa output yang dikembalikan?

  \item Apakah fungsi bergantung pada keadaan di luar dirinya?

\end{enumerate}

\begin{frame}[fragile]{Latihan akhir 1: menulis fungsi}
Tulis fungsi berikut dengan input dan output yang jelas.
\begin{enumerate}
  \item \texttt{polinom(x, koef)} untuk menghitung $a_0+a_1x+\cdots+a_nx^n$.
  \item \texttt{stat\_ringkas(a,b,c,d)} yang mengembalikan minimum, maksimum, rata-rata, dan rentang.

\end{enumerate}
\vspace{1mm}

\end{frame}
