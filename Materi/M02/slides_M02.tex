% 3
\begin{frame}{Apa itu subprogram?}
\begin{block}{Definisi kerja}
Subprogram adalah bagian program yang diberi nama untuk mengerjakan satu tugas tertentu. Subprogram dapat dipanggil dari bagian lain program tanpa menulis ulang langkah-langkahnya.
\end{block}
\vspace{1mm}
\begin{columns}[T]
\column{.47\textwidth}
\begin{block}{Dalam banyak bahasa}
\begin{itemize}
  \item \textbf{Prosedur}: mengerjakan aksi, sering tanpa nilai balik.
  \item \textbf{Fungsi}: mengolah input dan mengembalikan nilai.
\end{itemize}
\end{block}
\column{.49\textwidth}
\begin{block}{Dalam Python}
\begin{itemize}
  \item Keduanya ditulis dengan \texttt{def}.
  \item Jika tidak ada \texttt{return}, hasilnya \texttt{None}.
  \item Perbedaan utamanya ada pada tujuan pemakaian.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% 4
\begin{frame}{Makna fungsi dalam program}
\begin{center}
\includegraphics[width=.70\linewidth]{function_pipeline.png}
\end{center}
\vspace{-1mm}
\begin{block}{Cara membacanya}
Fungsi adalah unit bernama yang menerima masukan, menjalankan aturan tertentu, lalu menghasilkan keluaran. Dalam matematika kita sering menulis $y=f(x)$. Dalam program, fungsi juga dapat hanya menampilkan pesan, mengubah keadaan, atau memanggil fungsi lain.
\end{block}
\end{frame}

% 5
\begin{frame}[fragile]{Anatomi fungsi Python}
\begin{lstlisting}
def nama_fungsi(parameter_1, parameter_2):
    langkah_1
    langkah_2
    return hasil
\end{lstlisting}
\vspace{1mm}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Saat didefinisikan}
\begin{itemize}
  \item \texttt{def}: membuat definisi fungsi.
  \item Parameter: nama masukan di dalam fungsi.
  \item Badan fungsi: langkah-langkah yang dikerjakan.
\end{itemize}
\end{block}
\column{.48\textwidth}
\begin{block}{Saat dipanggil}
\begin{itemize}
  \item Argumen: nilai yang dikirim saat pemanggilan.
  \item \texttt{return}: nilai yang dikirim balik.
  \item Setelah \texttt{return}, eksekusi fungsi selesai.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Contoh pertama: fungsi dari rumus}
\begin{columns}[T]
\column{.43\textwidth}
\[
 p(x)=2-3x+x^2
\]
\[
 p(4)=2-12+16=6
\]
\column{.53\textwidth}
\begin{lstlisting}
def polinom_sederhana(x):
    return 2 - 3*x + x**2

hasil = polinom_sederhana(4)
print(hasil)
\end{lstlisting}
\end{columns}
\vspace{2mm}
\begin{block}{Perhatikan}
\texttt{x} pada definisi adalah parameter. Nilai \texttt{4} saat pemanggilan adalah argumen. Nilai yang dipakai pada operasi berikutnya berasal dari \texttt{return}.
\end{block}
\end{frame}

\begin{frame}[fragile]{Parameter dan argumen}
\begin{lstlisting}
def luas_persegi_panjang(panjang, lebar):

A = luas_persegi_panjang(5, 3)
B = luas_persegi_panjang(lebar=3, panjang=5)
\end{lstlisting}
\vspace{2mm}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Positional argument}
Urutan argumen penting. Nilai pertama masuk ke parameter pertama.
\end{block}
\column{.48\textwidth}
\begin{block}{Keyword argument}
Nama parameter disebut langsung. Urutan argumen menjadi lebih mudah dibaca.
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Empat bentuk fungsi: tanpa nilai balik}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Tanpa argumen, tanpa \texttt{return}}
\begin{lstlisting}
def salam():
    print("Selamat belajar")
\end{lstlisting}
Cocok untuk menampilkan pesan atau instruksi.
\end{block}
\column{.48\textwidth}
\begin{block}{Dengan argumen, tanpa \texttt{return}}
\begin{lstlisting}
def tampil_kuadrat(x):
    print(x**2)
\end{lstlisting}
Berguna jika hasil hanya ingin ditampilkan, bukan dihitung lanjut.
\end{block}
\end{columns}
\vspace{1mm}
\begin{block}{Catatan}
Jika tidak ada \texttt{return}, Python tetap mengembalikan \texttt{None}.
\end{block}
\end{frame}

\begin{frame}[fragile]{\texttt{print} tidak sama dengan \texttt{return}}
\begin{columns}[T]
\column{.48\textwidth}
\begin{lstlisting}
def g(x):
    return x**2 - 1

def h(x):
    print(g(x))
\end{lstlisting}
\column{.48\textwidth}
\begin{lstlisting}
y = h(3)
print(y)

z = h(3) + 2   # error
\end{lstlisting}
\end{columns}
\vspace{1mm}
\begin{block}{Yang terjadi}
\texttt{h(3)} menampilkan hasil \texttt{g(3)}, tetapi tidak mengembalikan nilai untuk dipakai pada operasi aritmatika. Karena itu \texttt{h(3)+2} gagal.
\end{block}
\end{frame}

% 11
\begin{frame}[fragile]{Fungsi yang mengembalikan lebih dari satu nilai}
\begin{lstlisting}
def stat_ringkas(a, b, c, d):
    nilai = [a, b, c, d]
    minimum = min(nilai)
    maksimum = max(nilai)
    rata = sum(nilai) / len(nilai)
    rentang = maksimum - minimum
    return minimum, maksimum, rata, rentang

mn, mx, r, rg = stat_ringkas(4, 7, 1, 9)
\end{lstlisting}
\vspace{1mm}
\begin{block}{Apa yang terjadi?}
Python mengembalikan satu objek bertipe tuple. Unpacking membagi isi tuple ke beberapa variabel.
\end{block}
\end{frame}

\begin{frame}[fragile]{Tuple dan unpacking}
\begin{columns}[T]
\column{.46\textwidth}

hasil = stat_ringkas(4, 7, 1, 9)
print(hasil)
print(type(hasil))
\end{lstlisting}
\column{.50\textwidth}
\begin{lstlisting}
minimum, maksimum, rata, rentang = hasil

# jumlah variabel harus cocok
x, y = hasil  # ValueError
\end{lstlisting}
\end{columns}
\vspace{1mm}
\begin{block}{Mengapa penting?}
Satu proses komputasi sering menghasilkan beberapa informasi: parameter, galat, jumlah iterasi, atau status konvergensi.
\end{block}
\end{frame}

% 13
\begin{frame}[fragile]{Scope: variabel lokal dan global}
\begin{lstlisting}
x = 10

def f(a):
    x = a + 2
    return x

print(f(5))
print(x)
\end{lstlisting}
\vspace{1mm}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Variabel lokal}
\texttt{x} di dalam fungsi hanya hidup di dalam fungsi tersebut.
\end{block}
\column{.48\textwidth}
\begin{block}{Variabel global}
\texttt{x} di luar fungsi tetap bernilai 10. Nilai ini tidak berubah hanya karena ada \texttt{x} lokal.
\end{block}
\end{columns}
\end{frame}

% 14
\begin{frame}[fragile]{Mengapa variabel global perlu hati-hati?}
\begin{lstlisting}
total = 0

def tambah(x):
    global total
    total = total + x
    return total
\end{lstlisting}
\vspace{1mm}
\begin{block}{Masalah yang sering muncul}
\begin{itemize}
  \item Hasil fungsi bergantung pada riwayat pemanggilan.
  \item Test menjadi tidak mandiri.
  \item Kesalahan sulit dilacak karena nilai dapat berubah dari tempat lain.
\end{itemize}
\end{block}
\begin{block}{Saran awal}
Utamakan fungsi yang menerima input secara eksplisit dan mengembalikan output secara eksplisit.
\end{block}
\end{frame}

E(p,q)=\frac{1}{1+d(p,q)^2}
\]
\[
d(p,q)=\sqrt{(p_1-q_1)^2+(p_2-q_2)^2}

def jarak(p, q):
    dx = p[0] - q[0]
    dy = p[1] - q[1]
    return (dx**2 + dy**2)**0.5

def energi(d):
    return 1 / (1 + d**2)

def energi_dua_titik(p, q):
    return energi(jarak(p, q))
\end{lstlisting}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Fungsi sebagai objek}
\begin{lstlisting}
def proses(x, f):
    return f(x)

def kuadrat(x):
    return x**2

def tambah1(x):
    return x + 1

print(proses(5, kuadrat))
print(proses(5, tambah1))
\end{lstlisting}
\vspace{1mm}
\begin{block}{Makna penting}
Nama \texttt{kuadrat} tanpa tanda kurung adalah objek fungsi. Objek itu dapat dikirim sebagai argumen ke fungsi lain.
\end{block}
\end{frame}

% 18
\begin{frame}[fragile]{List fungsi dan pipeline}
\begin{lstlisting}
def kuadrat(x):
    return x**2

def tambah1(x):
    return x + 1

def pipeline(x, ops):
    hasil = x
    for f in ops:
        hasil = f(hasil)
    return hasil

ops = [abs, kuadrat, tambah1]
print(pipeline(-3, ops))
\end{lstlisting}
\vspace{1mm}
\begin{block}{Baca alurnya}
\footnotesize Nilai $-3$ diproses oleh \texttt{abs}, lalu \texttt{kuadrat}, lalu \texttt{tambah1}. Ini mirip alur transformasi data.
\end{block}
\end{frame}

\begin{frame}[fragile]{Lambda: fungsi pendek untuk operasi pendek}
\begin{columns}[T]
\column{.45\textwidth}
\[
 f(x)=3x^2-2x+1
\]
\begin{lstlisting}
f = lambda x: 3*x**2 - 2*x + 1
\end{lstlisting}
\column{.51\textwidth}
\begin{lstlisting}
xnilai = range(-3, 4)
y = [f(x) for x in xnilai]

z = list(map(lambda x: 3*x**2 - 2*x + 1,
             xnilai))
\end{lstlisting}
\end{columns}
\vspace{1mm}
\begin{block}{Gunakan secara wajar}
Lambda cocok untuk fungsi satu baris yang jelas. Jika logika mulai panjang, fungsi bernama lebih baik.
\end{block}
\end{frame}

def f(x):

xnilai = [-3, -2, -1, 0, 1, 2, 3]

% 21
\begin{frame}{Rekursi: ide dasar}
\begin{block}{Definisi kerja}
Rekursi terjadi ketika sebuah fungsi memanggil dirinya sendiri untuk menyelesaikan masalah yang lebih kecil.
\end{block}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Dua bagian wajib}
\begin{itemize}
  \item \textbf{Base case}: kondisi berhenti.
  \item \textbf{Recursive case}: memanggil fungsi pada masalah yang lebih kecil.
\end{itemize}
\end{block}
\column{.48\textwidth}
\begin{block}{Pertanyaan saat membaca rekursi}
\begin{itemize}
  \item Apa yang menjadi ukuran masalah?
  \item Apakah ukuran itu berkurang?
  \item Kapan pemanggilan berhenti?
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% 22
\begin{frame}[fragile]{Contoh rekursi: faktorial}
\begin{columns}[T]
\column{.43\textwidth}
\[
0! = 1,\qquad n! = n(n-1)!
\]
\[
5! = 5\cdot 4\cdot 3\cdot 2\cdot 1
\]
\column{.53\textwidth}
\begin{lstlisting}
def faktorial(n):
    if n == 0:
        return 1
    return n * faktorial(n - 1)

print(faktorial(5))
\end{lstlisting}
\end{columns}
\vspace{1mm}
\begin{block}{Jejak singkat}
\footnotesize \texttt{faktorial(5)} memanggil \texttt{faktorial(4)}, lalu \texttt{faktorial(3)}, sampai mencapai \texttt{faktorial(0)}. Setelah base case tercapai, hasil dikalikan kembali ke atas.
\end{block}
\end{frame}

% 23
\begin{frame}[fragile]{Rekursi pada barisan}
\begin{columns}[T]
\column{.43\textwidth}
\[
a_0=2,\qquad a_n=3a_{n-1}-2
\]
\vspace{1mm}
Untuk menghitung $a_5$, fungsi perlu mengetahui $a_4$, $a_3$, dan seterusnya sampai $a_0$.
\column{.53\textwidth}
\begin{lstlisting}
def a(n):
    if n == 0:
        return 2
    return 3*a(n - 1) - 2

for k in range(6):
    print(k, a(k))
\end{lstlisting}
\end{columns}
\vspace{1mm}
\begin{block}{Catatan}
Rekursi cocok ketika definisi masalah memang bergantung pada versi yang lebih kecil dari dirinya sendiri.
\end{block}
\end{frame}

% 24
\begin{frame}[fragile]{Kesalahan umum pada rekursi}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Tidak ada base case}
\begin{lstlisting}
def turun(n):
    return turun(n - 1)
\end{lstlisting}
Fungsi terus memanggil dirinya sendiri.
\end{block}
\column{.48\textwidth}
\begin{block}{Tidak menuju base case}
\begin{lstlisting}
def salah(n):
    if n == 0:
        return 0
    return salah(n + 1)
\end{lstlisting}
Untuk $n>0$, pemanggilan makin jauh dari 0.
\end{block}
\end{columns}
\vspace{1mm}
\begin{block}{Hubungan dengan halt}
Sebuah fungsi rekursif harus bergerak menuju kondisi berhenti. Jika tidak, program tidak halt untuk input tertentu.
\end{block}
\end{frame}

% 25
\begin{frame}[fragile]{Merancang fungsi yang mudah diuji}
\begin{lstlisting}
def jarak_1d(x, y):
    return abs(x - y)

assert jarak_1d(3, 3) == 0
assert jarak_1d(3, 8) == 5
assert jarak_1d(-2, 3) == 5
\end{lstlisting}
\vspace{1mm}
\begin{block}{Kriteria awal}
\begin{itemize}
  \item Input jelas.
  \item Output jelas.
  \item Tidak bergantung pada variabel global.
  \item Dapat diuji dengan contoh yang jawabannya diketahui.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Checklist membaca fungsi}
\begin{block}{Saat melihat fungsi baru, tanyakan lima hal ini}
\begin{enumerate}
  \item Apa tugas yang dikerjakan fungsi?
  \item Apa input yang dibutuhkan?
  \item Apa output yang dikembalikan?
  \item Apakah fungsi hanya mencetak, atau benar-benar mengembalikan nilai?
  \item Apakah fungsi bergantung pada keadaan di luar dirinya?
\end{enumerate}
\end{block}

\end{block}
\end{frame}
