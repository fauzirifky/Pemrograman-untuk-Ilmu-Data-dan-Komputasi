% 5
\begin{frame}[fragile]{Anatomi fungsi Python}
\begin{lstlisting}
def nama_fungsi(parameter_1, parameter_2):
    langkah_1
    langkah_2
    return hasil
\end{lstlisting}
\vspace{2mm}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Bagian penting}
\begin{itemize}
  \item \texttt{def}: mendefinisikan fungsi.
  \item Nama fungsi: sebaiknya berupa kata kerja atau nama konsep.
  \item Parameter: nama variabel di dalam definisi fungsi.
\end{itemize}
\end{block}
\column{.48\textwidth}
\begin{block}{Saat dipanggil}
\begin{itemize}
  \item Argumen: nilai yang dikirim saat fungsi dipanggil.
  \item \texttt{return}: nilai yang dikirim kembali ke pemanggil.
  \item Fungsi selesai ketika mencapai \texttt{return}.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Contoh 1: fungsi dari rumus matematika}
\begin{columns}[T]
\column{.43\textwidth}
\[
 p(x)=2-3x+x^2
\]
\[
 p(4)=2-12+16=6
\]
\column{.53\textwidth}
\begin{lstlisting}
def polinom_sederhana(x):
    return 2 - 3*x + x**2

hasil = polinom_sederhana(4)
print(hasil)
\end{lstlisting}
\end{columns}
\vspace{2mm}
\begin{block}{Perhatikan}
Nilai \texttt{x} pada fungsi adalah parameter. Nilai \texttt{4} saat pemanggilan adalah argumen.
\end{block}
\end{frame}

\begin{frame}[fragile]{Parameter dan argumen}
\begin{lstlisting}
def luas_persegi_panjang(panjang, lebar):
    return panjang * lebar

A = luas_persegi_panjang(5, 3)
B = luas_persegi_panjang(lebar=3, panjang=5)
\end{lstlisting}
\vspace{2mm}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Positional argument}
Urutan argumen penting. Nilai pertama masuk ke parameter pertama.
\end{block}
\column{.48\textwidth}
\begin{block}{Keyword argument}
Nama parameter disebutkan langsung. Urutan tidak lagi menjadi fokus.
\end{block}
\end{columns}
\vspace{1mm}
\centering
\colorbox{LightBlue}{\parbox{.80\linewidth}{\centering Untuk fungsi matematis, nama parameter yang jelas mengurangi salah tafsir.}}
\end{frame}

% 8
\begin{frame}[fragile]{Empat bentuk fungsi}
\footnotesize
\begin{tabular}{p{3.5cm}p{3.8cm}p{4.6cm}}
\toprule
\textbf{Bentuk} & \textbf{Contoh} & \textbf{Kapan berguna} \\
\midrule
Tanpa argumen, tanpa \texttt{return} & \texttt{def salam(): print(...)} & Menampilkan pesan atau instruksi. \\
Tanpa argumen, dengan \texttt{return} & \texttt{def pi(): return 3.14159} & Menghasilkan konstanta atau konfigurasi. \\
Dengan argumen, tanpa \texttt{return} & \texttt{def tampil(x): print(x)} & Melaporkan hasil, bukan menghitung lanjut. \\
Dengan argumen, dengan \texttt{return} & \texttt{def f(x): return x**2} & Menghasilkan nilai untuk dipakai lagi. \\
\bottomrule
\end{tabular}
\vspace{2mm}
\begin{block}{Catatan}
Pada komputasi matematika, bentuk paling sering adalah: fungsi dengan argumen dan dengan \texttt{return}.
\end{block}
\end{frame}

% 9
\begin{frame}[fragile]{\texttt{print} tidak sama dengan \texttt{return}}
\begin{columns}[T]
\column{.48\textwidth}
\begin{lstlisting}
def g(x):
    return x**2 - 1

def h(x):
    print(g(x))
\end{lstlisting}
\column{.48\textwidth}
\begin{lstlisting}
y = h(3)
print(y)

z = h(3) + 2   # error
\end{lstlisting}
\end{columns}
\vspace{2mm}
\begin{block}{Ide utama}
\texttt{print} hanya menampilkan. Jika tidak ada \texttt{return}, fungsi mengembalikan \texttt{None}. Karena itu hasilnya tidak bisa langsung dipakai pada operasi aritmatika.
\end{block}
\end{frame}

% 10
\begin{frame}[fragile]{Fungsi yang mengembalikan lebih dari satu nilai}
\begin{lstlisting}
def stat_ringkas(a, b, c, d):
    nilai = [a, b, c, d]
    minimum = min(nilai)
    maksimum = max(nilai)
    rata = sum(nilai) / len(nilai)
    rentang = maksimum - minimum
    return minimum, maksimum, rata, rentang

mn, mx, r, rg = stat_ringkas(4, 7, 1, 9)
\end{lstlisting}
\vspace{1mm}
\begin{block}{Apa yang terjadi?}
Python sebenarnya mengembalikan satu objek bertipe tuple. Unpacking membagi isi tuple ke beberapa variabel.
\end{block}
\end{frame}

\begin{frame}[fragile]{Tuple dan unpacking}
\begin{columns}[T]
\column{.46\textwidth}
\begin{lstlisting}
hasil = stat_ringkas(4, 7, 1, 9)
print(hasil)
print(type(hasil))
\end{lstlisting}
\column{.50\textwidth}
\begin{lstlisting}
minimum, maksimum, rata, rentang = hasil

# jumlah variabel harus cocok
x, y = hasil  # ValueError
\end{lstlisting}
\end{columns}
\vspace{1mm}
\begin{block}{Mengapa penting?}
Dalam komputasi, satu proses sering menghasilkan beberapa informasi: parameter, galat, jumlah iterasi, atau status konvergensi.
\end{block}
\end{frame}

% 12
\begin{frame}[fragile]{Scope: variabel lokal dan global}
\begin{lstlisting}
x = 10

def f(a):
    x = a + 2
    return x

print(f(5))
print(x)
\end{lstlisting}
\vspace{2mm}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Variabel lokal}
\texttt{x} di dalam fungsi hanya hidup di dalam fungsi tersebut.
\end{block}
\column{.48\textwidth}
\begin{block}{Variabel global}
\texttt{x} di luar fungsi tetap bernilai 10. Nilai ini tidak otomatis berubah hanya karena ada \texttt{x} lokal.
\end{block}
\end{columns}
\end{frame}

% 13
\begin{frame}[fragile]{Mengapa variabel global perlu hati-hati?}
\begin{lstlisting}
total = 0

def tambah(x):
    global total
    total = total + x
    return total
\end{lstlisting}
\vspace{2mm}
\begin{block}{Masalah yang sering muncul}
\begin{itemize}
  \item Hasil fungsi bergantung pada riwayat pemanggilan.
  \item Test menjadi tidak mandiri.
  \item Kesalahan sulit dilacak karena nilai dapat berubah dari tempat lain.
\end{itemize}
\end{block}
\begin{block}{Saran awal}
Utamakan fungsi yang menerima input secara eksplisit dan mengembalikan output secara eksplisit.
\end{block}
\end{frame}

\begin{frame}[fragile]{Fungsi sebagai alat dekomposisi}
\begin{columns}[T]

\[
E(p,q)=\frac{1}{1+d(p,q)^2}

\[
d(p,q)=\sqrt{(p_1-q_1)^2+(p_2-q_2)^2}

def jarak(p, q):
    dx = p[0] - q[0]
    dy = p[1] - q[1]
    return (dx**2 + dy**2)**0.5

def energi(d):
    return 1 / (1 + d**2)

def energi_dua_titik(p, q):
    return energi(jarak(p, q))
\end{lstlisting}
\end{columns}

\begin{frame}[fragile]{Fungsi sebagai objek}
\begin{lstlisting}
def proses(x, f):
    return f(x)

def kuadrat(x):
    return x**2

def tambah1(x):
    return x + 1

print(proses(5, kuadrat))
print(proses(5, tambah1))
\end{lstlisting}
\vspace{1mm}
\begin{block}{Makna penting}
Nama \texttt{kuadrat} tanpa tanda kurung adalah objek fungsi. Objek itu dapat dikirim sebagai argumen ke fungsi lain.
\end{block}
\end{frame}

% 17
\begin{frame}[fragile]{List fungsi dan pipeline}
\begin{lstlisting}
def kuadrat(x):
    return x**2

def tambah1(x):
    return x + 1

def pipeline(x, ops):
    hasil = x
    for f in ops:
        hasil = f(hasil)
    return hasil

ops = [abs, kuadrat, tambah1]
print(pipeline(-3, ops))
\end{lstlisting}
\vspace{1mm}
\begin{block}{Baca alurnya}
\footnotesize Nilai $-3$ diproses oleh \texttt{abs}, lalu \texttt{kuadrat}, lalu \texttt{tambah1}. Ini mirip alur transformasi data.
\end{block}
\end{frame}

\begin{frame}[fragile]{Lambda: fungsi pendek untuk operasi pendek}
\begin{columns}[T]
\column{.45\textwidth}
\[
 f(x)=3x^2-2x+1
\]
\begin{lstlisting}
f = lambda x: 3*x**2 - 2*x + 1
\end{lstlisting}
\column{.51\textwidth}
\begin{lstlisting}
xnilai = range(-3, 4)
y = [f(x) for x in xnilai]

z = list(map(lambda x: 3*x**2 - 2*x + 1,
             xnilai))
\end{lstlisting}
\end{columns}
\vspace{1mm}
\begin{block}{Gunakan secara wajar}
Lambda cocok untuk fungsi satu baris yang jelas. Jika logika mulai panjang, fungsi bernama lebih baik.
\end{block}
\end{frame}

\begin{frame}[fragile]{List comprehension untuk evaluasi fungsi}
\begin{lstlisting}
def f(x):
    return 3*x**2 - 2*x + 1

xnilai = [-3, -2, -1, 0, 1, 2, 3]
y = [f(x) for x in xnilai]
\end{lstlisting}
\vspace{2mm}
\begin{block}{Cara membaca}
Untuk setiap $x$ di \texttt{xnilai}, hitung $f(x)$, lalu simpan hasilnya ke list baru.
\end{block}
\begin{block}{Batas materi}
Hari ini list comprehension dipakai secukupnya untuk memahami fungsi. Struktur data list dan array akan dibahas lebih kuat pada M3.
\end{block}
\end{frame}
